// SetTextureUniform sets a texture uniform value.
// Supported types: image.Image, *image.RGBA, image.Uniform
// The texture will be bound to an available texture unit.
// Note: Does NOT call Refresh() - caller should call Refresh() after all data is set up
func (s *Shader) SetTextureUniform(name string, value interface{}) {
	s.Textures[name] = value
	// Invalidate cache so the texture gets re-uploaded on next paint
	delete(s.textureCache, name)
}

// SetCubemapUniform sets a cubemap uniform value.
// Takes 6 images for the cubemap faces: [+X, -X, +Y, -Y, +Z, -Z]
// The cubemap will be bound to an available texture unit.
func (s *Shader) SetCubemapUniform(name string, faces [6]interface{}) {
	s.Cubemaps[name] = faces
	s.Refresh()
}

// SetCubemapUniformSilent sets a cubemap uniform without triggering Refresh.
// Caller must call Refresh() after all updates are done.
func (s *Shader) SetCubemapUniformSilent(name string, faces [6]interface{}) {
	s.Cubemaps[name] = faces
}

// SetVertices sets the vertex data for the shader.
// format should be one of: "pos2", "pos3", "pos3_norm3", "pos3_norm3_uv2", "pos2_uv2"
// or a custom format string describing the vertex layout.
func (s *Shader) SetVertices(data []float32, format string) {
	s.Vertices = data
	s.VertexFormat = format
	s.vertexCount = len(data) / s.attributeCountForFormat(format)
	s.Refresh()
}

// SetIndices sets the index buffer for indexed drawing (glDrawElements).
func (s *Shader) SetIndices(indices []uint16) {
	s.Indices = indices
	s.indexCount = len(indices)
	s.Refresh()
}

// SetIndicesNoRefresh sets the index buffer without triggering a Refresh.
// Use this when updating indices during command processing (before QueueDrawElements).
func (s *Shader) SetIndicesNoRefresh(indices []uint16) {
	s.Indices = indices
	s.indexCount = len(indices)
}

// attributeCountForFormat returns the number of floats per vertex for a given format.
func (s *Shader) attributeCountForFormat(format string) int {
	switch format {
	case "pos2":
		return 2
	case "pos3":
		return 3
	case "pos2_uv2":
		return 4 // 2 pos + 2 uv
	case "pos3_norm3":
		return 6 // 3 pos + 3 norm
	case "pos3_norm3_uv2":
		return 8 // 3 pos + 3 norm + 2 uv
	case "pos3_col4":
		return 7 // 3 pos + 4 color
	default:
		return 0 // Unknown format
	}
}

// SetAttributeBuffer sets data for a named vertex attribute (e.g., "position", "normal", "uv").
// size is the number of components per vertex (2 for vec2, 3 for vec3, 4 for vec4).
// stride is the byte stride between consecutive vertices (0 = tightly packed).
// offset is the byte offset into the buffer for this attribute.
func (s *Shader) SetAttributeBuffer(name string, data []float32, size int, stride int, offset int) {
	s.attrBuffersMu.Lock()
	defer s.attrBuffersMu.Unlock()
	if s.AttributeBuffers == nil {
		s.AttributeBuffers = make(map[string]*AttributeBuffer)
	}
	// Reuse existing *AttributeBuffer struct to avoid per-frame allocation
	if existing, ok := s.AttributeBuffers[name]; ok {
		// Only increment generation if the data actually changed.
		// pushAttribBuffersToShader calls this repeatedly with the same buffer.data
		// for shared geometry (e.g., 100 cubes using one BoxGeometry).
		// Comparing backing-array pointers detects whether glBufferData allocated new data.
		sameData := len(existing.Data) == len(data) && len(data) > 0 &&
			&existing.Data[0] == &data[0]
		if !sameData {
			existing.Generation++
		}
		existing.Data = data
		existing.Size = size
		existing.Stride = stride
		existing.Offset = offset
		// VBO is preserved (already set on the existing struct)
	} else {
		// Restore VBO from cache if available (saved by PruneAttributeBuffers)
		var cachedVBO uint32
		if s.vboCache != nil {
			cachedVBO = s.vboCache[name]
		}
		// Use monotonic counter so recreated entries with cached VBOs
		// always force a re-upload (VBOUploadedGen won't match).
		s.attrGenCounter++
		s.AttributeBuffers[name] = &AttributeBuffer{
			Data:       data,
			Size:       size,
			Stride:     stride,
			Offset:     offset,
			Generation: s.attrGenCounter,
			VBO:        cachedVBO,
		}
	}
	// Update vertex count based on the number of vertices in this buffer
	if size > 0 {
		s.vertexCount = len(data) / size
	}
}

// GetAttributeBuffer returns the buffer for a named attribute.
func (s *Shader) GetAttributeBuffer(name string) *AttributeBuffer {
	s.attrBuffersMu.RLock()
	defer s.attrBuffersMu.RUnlock()
	if s.AttributeBuffers == nil {
		return nil
	}
	return s.AttributeBuffers[name]
}

// GetAttributeBuffers returns a snapshot of all attribute buffers.
// The returned map is a shallow copy safe to iterate without holding the lock.
func (s *Shader) GetAttributeBuffers() map[string]*AttributeBuffer {
	s.attrBuffersMu.RLock()
	defer s.attrBuffersMu.RUnlock()
	if s.AttributeBuffers == nil {
		return nil
	}
	snapshot := make(map[string]*AttributeBuffer, len(s.AttributeBuffers))
	for k, v := range s.AttributeBuffers {
		snapshot[k] = v
	}
	return snapshot
}

// HasAttributeBuffers returns true if separate attribute buffers are set.
func (s *Shader) HasAttributeBuffers() bool {
	s.attrBuffersMu.RLock()
	defer s.attrBuffersMu.RUnlock()
	return s.AttributeBuffers != nil && len(s.AttributeBuffers) > 0
}

// PruneAttributeBuffers removes attribute buffers not in the keep set.
// Called before each draw to prevent stale attribute bindings from persisting
// when different objects use different sets of attributes.
// VBOs are preserved in vboCache so they can be reused when the attribute
// is re-added next frame (prevents VBO leak: prune deletes entry, painter
// writeback finds nil, next frame creates new VBO=0 → 11 VBOs/frame leaked).
func (s *Shader) PruneAttributeBuffers(keep map[string]bool) {
	s.attrBuffersMu.Lock()
	defer s.attrBuffersMu.Unlock()
	for name, buf := range s.AttributeBuffers {
		if !keep[name] {
			// Save VBO to cache before deleting so SetAttributeBuffer can reuse it
			if buf != nil && buf.VBO != 0 {
				if s.vboCache == nil {
					s.vboCache = make(map[string]uint32, 8)
				}
				s.vboCache[name] = buf.VBO
			}
			delete(s.AttributeBuffers, name)
		}
	}
}

// CacheVBO stores a VBO ID in the cache for an attribute name.
// Called by the painter after creating a VBO, so that future SetAttributeBuffer
// calls can reuse it even if PruneAttributeBuffers deleted the live entry.
func (s *Shader) CacheVBO(name string, vbo uint32) {
	s.attrBuffersMu.Lock()
	defer s.attrBuffersMu.Unlock()
	if s.vboCache == nil {
		s.vboCache = make(map[string]uint32, 8)
	}
	s.vboCache[name] = vbo
}

// ClearAttributeBufferVBOs clears VBO IDs (called on recompile to force re-upload).
func (s *Shader) ClearAttributeBufferVBOs() {
	s.attrBuffersMu.RLock()
	defer s.attrBuffersMu.RUnlock()
	for _, buf := range s.AttributeBuffers {
		buf.VBO = 0
	}
}

// SetAttributeLocation stores the location for a named attribute.
// This is set from the JS side and used instead of glGetAttribLocation
// to avoid issues with shader optimization removing unused attributes.
func (s *Shader) SetAttributeLocation(name string, location int32) {
	s.attribLocMu.Lock()
	defer s.attribLocMu.Unlock()
	if s.attributeLocations == nil {
		s.attributeLocations = make(map[string]int32)
	}
	s.attributeLocations[name] = location
}

// GetAttributeLocation returns the stored location for a named attribute.
// Returns -1 if not found.
func (s *Shader) GetAttributeLocation(name string) int32 {
	s.attribLocMu.RLock()
	defer s.attribLocMu.RUnlock()
	if s.attributeLocations == nil {
		return -1
	}
	loc, ok := s.attributeLocations[name]
	if !ok {
		return -1
	}
	return loc
}

// SetProgramCache stores a compiled program in the cache (for painter use).
func (s *Shader) SetProgramCache(key string, prog uint32) {
	if s.programCache == nil {
		s.programCache = make(map[string]uint32)
	}
	s.programCache[key] = prog
}

// AddPendingProgram queues shader sources for compilation by the painter.
func (s *Shader) AddPendingProgram(vertexSrc, fragmentSrc string) {
	s.pendingProgMu.Lock()
	s.pendingPrograms = append(s.pendingPrograms, PendingProgram{
		VertexSource:   vertexSrc,
		FragmentSource: fragmentSrc,
	})
	s.pendingProgMu.Unlock()
}

// DrainPendingPrograms returns and clears pending programs for compilation.
func (s *Shader) DrainPendingPrograms() []PendingProgram {
	s.pendingProgMu.Lock()
	defer s.pendingProgMu.Unlock()
	progs := s.pendingPrograms
	s.pendingPrograms = nil
	return progs
}

// DeleteTextureAllCaches removes a JS texture ID from ALL texture-related caches.
// Call this when a texture is deleted to prevent unbounded map growth.
func (s *Shader) DeleteTextureAllCaches(jsId int) {
	s.texMu.Lock()
	delete(s.gpuTexCache, jsId)
	delete(s.cpuTexImages, jsId)
	delete(s.gpuTexUploaded, jsId)
	delete(s.gpuTexFormats, jsId)
	// Also remove from jsTexForSampler (reverse lookup)
	for k, v := range s.jsTexForSampler {
		if v == jsId {
			delete(s.jsTexForSampler, k)
		}
	}
	s.texMu.Unlock()
}

func (s *Shader) DeleteVBOUploadedGen(vboId uint32) {
	delete(s.VBOUploadedGen, vboId)
}

// SetGPUTexUploaded marks a GPU texture as having been uploaded with real data.

// QueueDrawArraysInstanced queues an instanced drawArrays call with a snapshot of current geometry.
// Safe to share slice references: glBufferData always allocates new slices,
// and glBufferSubData uses copy-on-write.
func (s *Shader) QueueDrawArraysInstanced(mode uint32, first int, count int, instanceCount int) {
	s.attrBuffersMu.RLock()
	attrSnapshot := s.snapshotAttributeBuffers()
	s.attrBuffersMu.RUnlock()

	tag := s.CurrentDrawTag
	s.CurrentDrawTag = ""
	s.QueueRenderCommand(RenderCommand{
		Type: "drawArrays",
		Name: tag,
		Value: DrawArraysParams{
			Mode:             mode,
			First:            first,
			Count:            count,
			InstanceCount:    instanceCount,
			AttributeBuffers: attrSnapshot,
		},
	})
}

// QueueDrawElementsInstanced queues an instanced drawElements call with a snapshot of current geometry.
// Safe to share slice references: glBufferData always allocates new slices,
// and glBufferSubData uses copy-on-write.
func (s *Shader) QueueDrawElementsInstanced(mode uint32, count int, offset int, instanceCount int) {
	s.attrBuffersMu.RLock()
	attrSnapshot := s.snapshotAttributeBuffers()
	s.attrBuffersMu.RUnlock()

	tag := s.CurrentDrawTag
	s.CurrentDrawTag = ""
	s.QueueRenderCommand(RenderCommand{
		Type: "drawElements",
		Name: tag,
		Value: DrawElementsParams{
			Mode:             mode,
			Count:            count,
			Offset:           offset,
			InstanceCount:    instanceCount,
			AttributeBuffers: attrSnapshot,
			Indices:          s.Indices,
		},
	})
}
