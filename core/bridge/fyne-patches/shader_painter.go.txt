// shaderCacheKey computes a hash-based cache key for shader sources,
// avoiding the allocation of concatenating full source strings each frame.
func shaderCacheKey(vertexSrc, fragSrc string) string {
	h := fnv.New64a()
	h.Write([]byte(vertexSrc))
	h.Write([]byte{0}) // separator
	h.Write([]byte(fragSrc))
	return strconv.FormatUint(h.Sum64(), 36)
}

var shaderProfile = os.Getenv("TSYNE_SHADER_PROFILE") == "1"


// shaderAttribute represents a parsed vertex attribute declaration
type shaderAttribute struct {
	name           string
	glslType       string  // vec2, vec3, vec4, float, etc.
	size           int     // number of floats (2 for vec2, 3 for vec3, etc.)
	layoutLocation int     // layout(location=N) value, or -1 if not specified
}


// getDefaultShaderVertexSrc returns the default fullscreen quad vertex shader,
// using GLES 300 es syntax on ES backends and GLSL 110 on desktop.
func getDefaultShaderVertexSrc() string {
	if isGLESBackend {
		return `#version 300 es
precision highp float;
in vec2 vert;
out vec2 v_texCoord;
void main() {
  v_texCoord = vert * 0.5 + 0.5;
  gl_Position = vec4(vert, 0.0, 1.0);
}
`
	}
	return `#version 110
attribute vec2 vert;
varying vec2 v_texCoord;
void main() {
  v_texCoord = vert * 0.5 + 0.5;
  gl_Position = vec4(vert, 0.0, 1.0);
}
`
}

// getShaderFragmentPrefix returns the standard uniform prefix for cosyne-style
// fragment shaders, using GLES 300 es syntax on ES backends and GLSL 110 on desktop.
func getShaderFragmentPrefix() string {
	if isGLESBackend {
		return `#version 300 es
precision highp float;
uniform vec2 u_resolution;
uniform float u_time;
in vec2 v_texCoord;
out vec4 fragColor;
#define gl_FragColor fragColor
`
	}
	return `#version 110
uniform vec2 u_resolution;
uniform float u_time;
varying vec2 v_texCoord;
`
}

// Always on — the cost is negligible (O(indexCount) scan per draw) and
// a skipped draw is infinitely better than a frozen machine.

// validateDrawElements checks that a drawElements call won't read out-of-bounds.
// Returns true if safe to draw, false if the call should be skipped.
func validateDrawElements(indices []uint16, count, byteOffset int, maxVertexCount int, instanceCount int, cmdIdx int, prog Program) bool {
	// Check 0: must have indices to draw indexed geometry
	if len(indices) == 0 {
		log.Printf("[GL_VALIDATE] BLOCKED drawElements cmd#%d prog=%d: empty index buffer", cmdIdx, prog)
		return false
	}

	idxOffset := byteOffset / 2 // byte offset → index offset (uint16 = 2 bytes)
	if idxOffset < 0 {
		idxOffset = 0
	}

	// Check 1: index buffer bounds
	if idxOffset+count > len(indices) {
		log.Printf("[GL_VALIDATE] BLOCKED drawElements cmd#%d prog=%d: index overrun idxOffset=%d count=%d indexBufLen=%d",
			cmdIdx, prog, idxOffset, count, len(indices))
		return false
	}

	// Check 2: scan drawn indices for max value vs vertex count
	maxIdx := uint16(0)
	for i := idxOffset; i < idxOffset+count; i++ {
		if indices[i] > maxIdx {
			maxIdx = indices[i]
		}
	}
	if maxVertexCount > 0 && int(maxIdx) >= maxVertexCount {
		log.Printf("[GL_VALIDATE] BLOCKED drawElements cmd#%d prog=%d: maxIndex=%d >= maxVertexCount=%d (inst=%d)",
			cmdIdx, prog, maxIdx, maxVertexCount, instanceCount)
		return false
	}
	// If we have no vertex count info at all (instanced-only), cap the max index
	// at a conservative limit to prevent GPU from reading into unmapped memory.
	// 100K is generous enough for any real geometry but catches runaway reads.
	if maxVertexCount == 0 && int(maxIdx) > 100_000 {
		log.Printf("[GL_VALIDATE] BLOCKED drawElements cmd#%d prog=%d: maxIndex=%d with unknown vertex count (inst=%d)",
			cmdIdx, prog, maxIdx, instanceCount)
		return false
	}

	// Check 3: sanity check count
	if count > 10_000_000 {
		log.Printf("[GL_VALIDATE] BLOCKED drawElements cmd#%d prog=%d: suspicious count=%d", cmdIdx, prog, count)
		return false
	}

	return true
}

// validateDrawArrays checks that a drawArrays call won't read out-of-bounds.
func validateDrawArrays(first, count, maxVertexCount int, instanceCount int, cmdIdx int, prog Program) bool {
	if maxVertexCount > 0 && first+count > maxVertexCount {
		log.Printf("[GL_VALIDATE] BLOCKED drawArrays cmd#%d prog=%d: first=%d+count=%d > maxVertexCount=%d (inst=%d)",
			cmdIdx, prog, first, count, maxVertexCount, instanceCount)
		return false
	}
	// If no vertex count info, cap at a conservative limit
	if maxVertexCount == 0 && count > 100_000 {
		log.Printf("[GL_VALIDATE] BLOCKED drawArrays cmd#%d prog=%d: count=%d with unknown vertex count (inst=%d)",
			cmdIdx, prog, count, instanceCount)
		return false
	}
	if count > 10_000_000 {
		log.Printf("[GL_VALIDATE] BLOCKED drawArrays cmd#%d prog=%d: suspicious count=%d", cmdIdx, prog, count)
		return false
	}
	return true
}

// getMaxIndex scans drawn indices and returns the maximum index value.
func getMaxIndex(indices []uint16, count, byteOffset int) int {
	idxOffset := byteOffset / 2
	if idxOffset < 0 {
		idxOffset = 0
	}
	maxIdx := 0
	end := idxOffset + count
	if end > len(indices) {
		end = len(indices)
	}
	for i := idxOffset; i < end; i++ {
		if int(indices[i]) > maxIdx {
			maxIdx = int(indices[i])
		}
	}
	return maxIdx
}

// validateBufferAccessForDraw checks that no attribute buffer would be overread
// by a draw call with the given vertex/instance counts.
// maxVertexIdx is the highest vertex index accessed (count-1 for arrays, maxIndex for elements).
// Returns true if safe, false if any buffer would be overread.
func validateBufferAccessForDraw(attrBuffers map[string]*canvas.AttributeBuffer, attribDivisors map[string]uint32, maxVertexIdx int, instanceCount int, cmdIdx int, prog Program) bool {
	for name, buf := range attrBuffers {
		if buf == nil || len(buf.Data) == 0 || buf.Size <= 0 {
			continue
		}
		bufByteSize := len(buf.Data) * 4
		stride := buf.Stride
		if stride == 0 {
			stride = buf.Size * 4 // tightly packed
		}

		div := uint32(0)
		if attribDivisors != nil {
			div = attribDivisors[name]
		}

		var maxIdx int
		if div > 0 && instanceCount > 0 {
			maxIdx = (instanceCount - 1) / int(div)
		} else if div > 0 {
			maxIdx = 0
		} else {
			maxIdx = maxVertexIdx
		}
		if maxIdx < 0 {
			maxIdx = 0
		}

		// Last byte accessed: maxIdx * stride + offset + componentSize
		lastByte := maxIdx*stride + buf.Offset + buf.Size*4
		if lastByte > bufByteSize {
			log.Printf("[GL_DIAG] OVERREAD buffer '%s' cmd#%d prog=%d: maxIdx=%d stride=%d offset=%d size*4=%d => lastByte=%d > bufSize=%d (inst=%d div=%d)",
				name, cmdIdx, prog, maxIdx, stride, buf.Offset, buf.Size*4, lastByte, bufByteSize, instanceCount, div)
			return false
		}
		if buf.Offset < 0 {
			log.Printf("[GL_DIAG] NEGATIVE OFFSET buffer '%s' cmd#%d prog=%d: offset=%d",
				name, cmdIdx, prog, buf.Offset)
			return false
		}
	}
	return true
}

// logDrawCall logs full parameters for a draw call (for TSYNE_GL_DIAG diagnosis).
func logDrawCall(drawType string, cmdIdx int, prog Program, mode uint32, first int, count int, offset int, instanceCount int,
	attrBuffers map[string]*canvas.AttributeBuffer, attribDivisors map[string]uint32, indices []uint16) {
	log.Printf("[GL_DIAG] >>> %s cmd#%d prog=%d mode=%d first=%d count=%d offset=%d inst=%d",
		drawType, cmdIdx, prog, mode, first, count, offset, instanceCount)
	for name, buf := range attrBuffers {
		if buf == nil {
			continue
		}
		stride := buf.Stride
		if stride == 0 {
			stride = buf.Size * 4
		}
		div := uint32(0)
		if attribDivisors != nil {
			div = attribDivisors[name]
		}
		log.Printf("[GL_DIAG]   attr '%s': dataLen=%d size=%d stride=%d offset=%d VBO=%d gen=%d div=%d",
			name, len(buf.Data), buf.Size, buf.Stride, buf.Offset, buf.VBO, buf.Generation, div)
	}
	if len(indices) > 0 {
		maxIdx := getMaxIndex(indices, count, offset)
		preview := 6
		if len(indices) < preview {
			preview = len(indices)
		}
		log.Printf("[GL_DIAG]   indices: len=%d maxIdx=%d first%d=%v", len(indices), maxIdx, preview, indices[:preview])
	}
}

// glErrorCount tracks total GL errors for rate-limiting log output
var glErrorCount int64

// checkGLError logs any pending GL error after a draw call.
// Drains ALL pending errors (GL can queue multiple).
func (p *painter) checkGLError(drawType string, cmdIdx int, prog Program) {
	for {
		glErr := p.ctx.GetError()
		if glErr == 0 {
			break
		}
		glErrorCount++
		// Log first 50, then every 100th
		if glErrorCount <= 50 || glErrorCount%100 == 0 {
			errName := "UNKNOWN"
			switch glErr {
			case 0x0500:
				errName = "INVALID_ENUM"
			case 0x0501:
				errName = "INVALID_VALUE"
			case 0x0502:
				errName = "INVALID_OPERATION"
			case 0x0505:
				errName = "OUT_OF_MEMORY"
			case 0x0506:
				errName = "INVALID_FRAMEBUFFER_OPERATION"
			}
			log.Printf("[GL_VALIDATE] GL ERROR 0x%04X (%s) after %s cmd#%d prog=%d (total errors: %d)",
				glErr, errName, drawType, cmdIdx, prog, glErrorCount)
		}
	}
}

// getMaxVertexCount finds the maximum vertex count from any non-instanced attribute buffer.
// Tries "position" first, then falls back to any buffer — this prevents the case where
// minified attribute names cause a "position" lookup miss, returning 0 and disabling
// all bounds checking.
func getMaxVertexCount(attrBuffers map[string]*canvas.AttributeBuffer, attribDivisors map[string]uint32) int {
	// Fast path: check "position" buffer directly
	if buf := attrBuffers["position"]; buf != nil && buf.Size > 0 && len(buf.Data) > 0 {
		return len(buf.Data) / buf.Size
	}
	// Slow path: find the largest non-instanced buffer as our vertex count bound.
	// If all buffers are instanced (divisor > 0), return 0 (instanced-only draws).
	maxVtx := 0
	for name, buf := range attrBuffers {
		if buf == nil || buf.Size <= 0 || len(buf.Data) == 0 {
			continue
		}
		if attribDivisors != nil && attribDivisors[name] > 0 {
			continue // Skip instanced buffers
		}
		vtx := len(buf.Data) / buf.Size
		if vtx > maxVtx {
			maxVtx = vtx
		}
	}
	return maxVtx
}

// profileAccum tracks cumulative timing across multiple drawShader calls
var profileAccum struct {
	count       int
	compileNs   int64
	attribNs    int64
	textureNs   int64
	cmdLoopNs   int64
	uniformNs   int64 // time in uniform commands specifically
	uniformCount int

}

// drawShader renders a Shader canvas object
func (p *painter) drawShader(shader *canvas.Shader, pos fyne.Position, frame fyne.Size) {
	// Always signal paint done, even on early returns.
	// Without this, WaitForPaint blocks for 5s on every frame when geometry
	// isn't ready yet (race on ~1-in-10 launches), permanently stuck at 1fps.
	defer shader.SignalPaintDone()

	if shader.FragmentSource == "" {
		return
	}

	var profileStart, phaseStart time.Time
	if shaderProfile {
		profileStart = time.Now()
		phaseStart = profileStart
	}

	// Reset per-frame GPU resource creation counters
	paintVBOCreates = 0
	paintIBOCreates = 0
	paintTexCreates = 0
	paintFBOCreates = 0
	paintRBOCreates = 0
	paintCounter++
	paintStart := time.Now()

	// If shader has a custom vertex source (3D/geometry mode), wait for attribute data
	// This prevents rendering a white rectangle before geometry data arrives
	if shader.VertexSource != "" && !shader.HasAttributeBuffers() && shader.GetVertexCount() == 0 {
		// Custom vertex shader but no geometry data yet - skip rendering
		return
	}

	// Compile shader if needed
	// Check both needsCompile flag AND program cache to detect source changes.
	// glUseProgram uses direct field assignment (no needsCompile) to avoid
	// redundant recompilation in multi-program scenes. We detect the need to
	// compile by checking if the current sources have a cached program.
	vertexSrc := shader.VertexSource
	if vertexSrc == "" {
		vertexSrc = getDefaultShaderVertexSrc()
	}
	cacheKey := shaderCacheKey(vertexSrc, shader.FragmentSource)
	cachedProg, hasCachedProg := shader.GetProgramCache()[cacheKey]
	needsCompile := shader.NeedsCompile()

	if hasCachedProg && cachedProg != 0 && !needsCompile {
		// Sources match a cached program - use it without recompiling
		if shader.Program() != cachedProg {
			shader.SetProgram(cachedProg)
			// Clear uniform location cache when switching programs
			for k := range shader.UniformLocs() {
				delete(shader.UniformLocs(), k)
			}
		}
	} else if needsCompile || shader.Program() == 0 || !hasCachedProg {
		prog := p.compileShaderProgram(vertexSrc, shader.FragmentSource)
		if prog == 0 {
			return
		}
		shader.SetProgram(uint32(prog))
		// Cache program so subsequent frames and useProgram commands get cache hits
		shader.SetProgramCache(cacheKey, uint32(prog))
		// Clear uniform location cache on recompile
		for k := range shader.UniformLocs() {
			delete(shader.UniformLocs(), k)
		}
	}

	prog := Program(shader.Program())
	if prog == 0 {
		return
	}
	if shaderDebug {
		log.Printf("[drawShader] Using program %d (vertexSrc=%d bytes, fragSrc=%d bytes)", prog, len(vertexSrc), len(shader.FragmentSource))
		if len(shader.FragmentSource) > 10000 {
			os.WriteFile("/tmp/tsyne_frag.glsl", []byte(shader.FragmentSource), 0644)
			os.WriteFile("/tmp/tsyne_vert.glsl", []byte(vertexSrc), 0644)
		}
	}

	// Save current program
	p.ctx.UseProgram(prog)

	// Get or cache uniform locations
	getUniformLoc := func(name string) Uniform {
		if loc, ok := shader.UniformLocs()[name]; ok {
			return Uniform(loc)
		}
		loc := p.ctx.GetUniformLocation(prog, name)
		shader.SetUniformLoc(name, int32(loc))
		return loc
	}

	// Calculate pixel position for uniforms
	// Note: OpenGL has origin at bottom-left, Fyne has origin at top-left
	pixelX := p.pixScale * pos.X
	// For gl_FragCoord, y=0 is at bottom of window
	// Shader bottom edge in Fyne coords: pos.Y + shader.Size().Height
	// In OpenGL coords: frame.Height - (pos.Y + shader.Size().Height)
	pixelY := p.pixScale * (frame.Height - pos.Y - shader.Size().Height)
	pixelWidth := p.pixScale * shader.Size().Width
	pixelHeight := p.pixScale * shader.Size().Height

	// Set standard uniforms
	resLoc := getUniformLoc("u_resolution")
	if resLoc >= 0 {
		p.ctx.Uniform2f(resLoc, pixelWidth, pixelHeight)
	}

	// Set viewport to the shader's area so gl_FragCoord is relative to it
	// This makes the shader's coordinate system match what it expects
	viewportX := int(pixelX)
	viewportY := int(pixelY)
	viewportW := int(pixelWidth)
	viewportH := int(pixelHeight)
	if shaderDebug {
		log.Printf("[drawShader] viewport: x=%d y=%d w=%d h=%d (pixScale=%.2f, shaderSize=%.0fx%.0f, frameSize=%.0fx%.0f, pos=%.0f,%.0f)",
			viewportX, viewportY, viewportW, viewportH, p.pixScale,
			shader.Size().Width, shader.Size().Height, frame.Width, frame.Height, pos.X, pos.Y)
	}
	p.ctx.Viewport(viewportX, viewportY, viewportW, viewportH)

	// For 3D shaders: pre-clear the COLOR buffer only with the stored
	// clear color.  Fyne's painter.Clear() clears the entire framebuffer
	// to the theme background (white) before painting widgets.  This
	// restores the shader's region to the correct scene background so
	// white never bleeds through.  Only clear color — depth is cleared
	// by the render commands' own clear call.
	if shader.VertexSource != "" {
		cc := shader.GetClearColor()
		p.ctx.ClearColor(cc[0], cc[1], cc[2], cc[3])
		p.ctx.Enable(scissorTest)
		p.ctx.Scissor(int32(viewportX), int32(viewportY), int32(viewportW), int32(viewportH))
		p.ctx.Clear(colorBufferBit)
		p.ctx.Disable(scissorTest)
	}

	// With viewport set, u_position should be 0,0 since gl_FragCoord is now relative to viewport
	posLoc := getUniformLoc("u_position")
	if posLoc >= 0 {
		p.ctx.Uniform2f(posLoc, 0, 0)
	}

	timeLoc := getUniformLoc("u_time")
	if timeLoc >= 0 {
		// Use the shader's cached time - only updated when Refresh() is called
		// This matches Fyne's texture caching behavior where widgets only update on explicit refresh
		p.ctx.Uniform1f(timeLoc, shader.CachedTime())
	}

	// Helper function to set a uniform value
	setUniform := func(name string, val interface{}) {
		loc := getUniformLoc(name)
		if loc < 0 {
			return
		}
		switch v := val.(type) {
		case float32:
			p.ctx.Uniform1f(loc, v)
		case float64:
			p.ctx.Uniform1f(loc, float32(v))
		case int:
			p.ctx.Uniform1f(loc, float32(v))
		case int8:
			p.ctx.Uniform1f(loc, float32(v))
		case int16:
			p.ctx.Uniform1f(loc, float32(v))
		case int32:
			p.ctx.Uniform1i(loc, v)
		case int64:
			p.ctx.Uniform1f(loc, float32(v))
		case uint8:
			p.ctx.Uniform1f(loc, float32(v))
		case uint16:
			p.ctx.Uniform1f(loc, float32(v))
		case uint32:
			p.ctx.Uniform1f(loc, float32(v))
		case uint64:
			p.ctx.Uniform1f(loc, float32(v))
		case []int32:
			if len(v) >= 1 {
				p.ctx.Uniform1i(loc, v[0])
			}
		case [2]float32:
			p.ctx.Uniform2f(loc, v[0], v[1])
		case [3]float32:
			p.ctx.Uniform3f(loc, v[0], v[1], v[2])
		case [4]float32:
			p.ctx.Uniform4f(loc, v[0], v[1], v[2], v[3])
		case [9]float32:
			p.ctx.UniformMatrix3fv(loc, false, v[:])
		case [16]float32:
			if shaderDebug {
				log.Printf("[drawShader] uniform mat4 %s = [%.4f %.4f %.4f %.4f / ...] loc=%d",
					name, v[0], v[1], v[2], v[3], loc)
			}
			p.ctx.UniformMatrix4fv(loc, false, v[:])
		case []float32:
			switch len(v) {
			case 1:
				p.ctx.Uniform1f(loc, v[0])
			case 2:
				p.ctx.Uniform2f(loc, v[0], v[1])
			case 3:
				if shaderDebug {
					log.Printf("[drawShader] uniform vec3 %s = (%.4f, %.4f, %.4f) loc=%d", name, v[0], v[1], v[2], loc)
				}
				p.ctx.Uniform3f(loc, v[0], v[1], v[2])
			case 4:
				p.ctx.Uniform4f(loc, v[0], v[1], v[2], v[3])
			case 9:
				p.ctx.UniformMatrix3fv(loc, false, v)
			case 16:
				if shaderDebug {
					log.Printf("[drawShader] uniform mat4 %s = [%.4f %.4f %.4f %.4f / %.4f %.4f %.4f %.4f / %.4f %.4f %.4f %.4f / %.4f %.4f %.4f %.4f] loc=%d",
						name, v[0], v[1], v[2], v[3], v[4], v[5], v[6], v[7], v[8], v[9], v[10], v[11], v[12], v[13], v[14], v[15], loc)
				}
				p.ctx.UniformMatrix4fv(loc, false, v)
			}
		case []interface{}:
			floats := make([]float32, len(v))
			for i, x := range v {
				switch f := x.(type) {
				case float64:
					floats[i] = float32(f)
				case float32:
					floats[i] = f
				case int:
					floats[i] = float32(f)
				case int8:
					floats[i] = float32(f)
				case int16:
					floats[i] = float32(f)
				case int32:
					floats[i] = float32(f)
				case int64:
					floats[i] = float32(f)
				case uint8:
					floats[i] = float32(f)
				case uint16:
					floats[i] = float32(f)
				case uint32:
					floats[i] = float32(f)
				case uint64:
					floats[i] = float32(f)
				}
			}
			switch len(floats) {
			case 1:
				p.ctx.Uniform1f(loc, floats[0])
			case 2:
				p.ctx.Uniform2f(loc, floats[0], floats[1])
			case 3:
				p.ctx.Uniform3f(loc, floats[0], floats[1], floats[2])
			case 4:
				p.ctx.Uniform4f(loc, floats[0], floats[1], floats[2], floats[3])
			}
		}
	}

	// Set user-defined uniforms (e.g. u_center, u_zoom, u_maxIter for fractals)
	for name, val := range shader.GetUniformsCopy() {
		setUniform(name, val)
	}

	if shaderProfile {
		now := time.Now()
		profileAccum.compileNs += now.Sub(phaseStart).Nanoseconds()
		phaseStart = now
	}

	// Skip Phase 2.1/2.2 (legacy texture setup) when render commands handle
	// texture binding.  Three.js scenes use activeTexture/bindTexture render
	// commands to bind different textures per draw call.  Phase 2.1 can
	// interfere: it creates duplicate GL textures, overwrites gpuTexCache
	// entries, and races with the JS thread writing to shader.Textures.
	skipLegacyTextures := shader.HasRenderHistory()

	// Set texture uniforms (Phase 2.1)
	// Upload texture data and bind to sampler uniforms
	textureUnit := 0
	if !skipLegacyTextures {
	textures := shader.GetTextures()
	for texName, texData := range textures {
		if textureUnit >= 16 {
			break
		}

		// Get or create GL texture
		texID, cached := shader.GetTextureCache()[texName]
		if !cached {
			tex := p.ctx.CreateTexture()
			paintTexCreates++
			texID = uint32(tex)
			shader.SetTextureCache(texName, texID)
		}

		// Activate texture unit and bind texture
		p.ctx.ActiveTexture(texture0 + uint32(textureUnit))
		p.ctx.BindTexture(texture2D, Texture(texID))

		// Upload texture data if not cached (or always upload for now)
		if !cached {
			// Set texture parameters
			p.ctx.TexParameteri(texture2D, textureMinFilter, linear)
			p.ctx.TexParameteri(texture2D, textureMagFilter, linear)
			p.ctx.TexParameteri(texture2D, textureWrapS, clampToEdge)
			p.ctx.TexParameteri(texture2D, textureWrapT, clampToEdge)

			// Convert texture data to RGBA
			rgba := p.textureToRGBA(texData)
			if rgba != nil && len(rgba.Pix) > 0 {
				// Check if this texture has a JS ID with RED format info
				jsTexId := shader.GetJSTexForSampler(texName)
				texFmt := shader.GetGPUTexFormat(jsTexId)
				w := rgba.Rect.Size().X
				h := rgba.Rect.Size().Y
				if texFmt.Format == 0x1903 { // GL_RED
					redData := make([]byte, w*h)
					for i := 0; i < w*h; i++ {
						redData[i] = rgba.Pix[i*4]
					}
					p.ctx.TexImage2D(texture2D, 0, w, h, texFmt.Format, unsignedByte, redData)
				} else {
					p.ctx.TexImage2D(texture2D, 0, w, h, colorFormatRGBA, unsignedByte, rgba.Pix)
				}
			}
		}

		// Set sampler uniform to texture unit
		shader.SetTextureUnit(texName, textureUnit)
		samplerLoc := getUniformLoc(texName)
		if samplerLoc >= 0 {
			p.ctx.Uniform1i(samplerLoc, int32(textureUnit))
		}

		// If this sampler has a JS texture ID mapping, populate gpuTexCache
		// so that bindTexture render commands can find this uploaded texture
		// (overwriting the empty GPU texture created by texImage2D_gpu/texStorage2D)
		if jsTexId := shader.GetJSTexForSampler(texName); jsTexId != 0 {
			shader.SetGPUTexCache(jsTexId, texID)
		}

		textureUnit++
	}

	// Set cubemap uniforms (Phase 2.2)
	// Full cubemap texture upload and binding implementation
	for cubemapName, faces := range shader.GetCubemaps() {
		if textureUnit >= 16 {
			log.Printf("[drawShader] Warning: too many textures+cubemaps (max 16), skipping %s", cubemapName)
			break
		}

		// Get or create cubemap texture
		cubemapTexID, cached := shader.GetCubemapCache()[cubemapName]
		if !cached {
			// Create new cubemap texture
			tex := p.ctx.CreateTexture()
			paintTexCreates++
			cubemapTexID = uint32(tex)
			shader.SetCubemapCache(cubemapName, cubemapTexID)
		}

		// Activate texture unit and bind cubemap
		p.ctx.ActiveTexture(texture0 + uint32(textureUnit))
		p.ctx.BindTexture(textureCube, Texture(cubemapTexID))

		// Upload faces if not cached (or always upload for now - caching optimization later)
		if !cached {
			// Set cubemap texture parameters
			p.ctx.TexParameteri(textureCube, textureMinFilter, linear)
			p.ctx.TexParameteri(textureCube, textureMagFilter, linear)
			p.ctx.TexParameteri(textureCube, textureWrapS, clampToEdge)
			p.ctx.TexParameteri(textureCube, textureWrapT, clampToEdge)

			// Upload each of the 6 faces
			faceTargets := []uint32{
				textureCubePositiveX, // +X (right)
				textureCubeNegativeX, // -X (left)
				textureCubePositiveY, // +Y (up)
				textureCubeNegativeY, // -Y (down)
				textureCubePositiveZ, // +Z (front)
				textureCubeNegativeZ, // -Z (back)
			}

			for i, faceTarget := range faceTargets {
				faceData := faces[i]
				if faceData == nil {
					continue
				}

				// Convert face data to RGBA pixel data
				rgba := p.cubemapFaceToRGBA(faceData)
				if rgba == nil || len(rgba.Pix) == 0 {
					log.Printf("[drawShader] Warning: empty face data for cubemap %s face %d", cubemapName, i)
					continue
				}

				// Upload face to cubemap
				p.ctx.TexImage2D(
					faceTarget,
					0,
					rgba.Rect.Size().X,
					rgba.Rect.Size().Y,
					colorFormatRGBA,
					unsignedByte,
					rgba.Pix,
				)
			}
			p.logError()
		}

		// Set samplerCube uniform to texture unit number
		shader.SetCubemapUnit(cubemapName, textureUnit)
		samplerLoc := getUniformLoc(cubemapName)
		if samplerLoc >= 0 {
			p.ctx.Uniform1i(samplerLoc, int32(textureUnit))
		}
		textureUnit++
	}
	} // end if !skipLegacyTextures

	if shaderProfile {
		now := time.Now()
		profileAccum.textureNs += now.Sub(phaseStart).Nanoseconds()
		phaseStart = now
	}

	// Parse vertex shader to find attributes dynamically
	// (vertexSrc was already computed above for the compilation check)
	attrs := parseVertexAttributes(vertexSrc)
	// if shaderDebug {
	// 	log.Printf("[drawShader] Parsed %d attributes from vertex shader", len(attrs))
	// 	for _, a := range attrs {
	// 		log.Printf("[drawShader]   parsed attr: %s (%s, size=%d)", a.name, a.glslType, a.size)
	// 	}
	// }

	// Build a set of GL locations reserved by mat4 attributes.
	// mat4 attributes occupy 4 consecutive locations (one per column).
	// We must prevent the stored-location fallback from binding other
	// attributes to locations that overlap with mat4 columns.
	// Reuse shader.Mat4Locs to avoid per-frame map allocation.
	mat4Locs := shader.Mat4Locs
	if mat4Locs == nil {
		mat4Locs = make(map[int]bool)
		shader.Mat4Locs = mat4Locs
	} else {
		for k := range mat4Locs {
			delete(mat4Locs, k)
		}
	}
	for _, a := range attrs {
		if a.glslType == "mat4" {
			glLoc := p.ctx.GetAttribLocation(prog, a.name)
			if glLoc >= 0 {
				for col := 0; col < 4; col++ {
					mat4Locs[int(glLoc)+col] = true
				}
			}
		}
	}

	// Track enabled attributes for cleanup (pre-allocate to reduce GC pressure)
	enabledAttribs := make([]Attribute, 0, 16)

	// Create a VAO for this shader draw.
	// On modern GL contexts (3.0+), vertex attribute state (including divisors)
	// is per-VAO. Without a bound VAO, VertexAttribDivisor may not take effect.
	shaderVAO := p.ctx.GenVertexArray()
	p.ctx.BindVertexArray(shaderVAO)

	// Check if we have separate attribute buffers (three.js style)
	useSeparateBuffers := shader.HasAttributeBuffers()
	// When render commands exist, the command loop handles all VBO/attrib
	// setup per-draw. Phase 1's GL calls (CreateBuffer, BindBuffer,
	// BufferData, EnableVertexAttribArray, VertexAttribPointer) are
	// completely redundant — they create VBOs that the command loop
	// never uses, and EnableVertexAttribArray calls that the command
	// loop overrides. Skip them to avoid wasted GL traffic.
	hasRenderCmds := shader.HasRenderHistory()
	// if shaderDebug {
	// 	log.Printf("[drawShader] useSeparateBuffers=%v, HasAttributeBuffers=%v", useSeparateBuffers, shader.HasAttributeBuffers())
	// 	if shader.AttributeBuffers != nil {
	// 		log.Printf("[drawShader] AttributeBuffers count: %d", len(shader.AttributeBuffers))
	// 		for name, buf := range shader.AttributeBuffers {
	// 			if buf != nil {
	// 				log.Printf("[drawShader]   attr %s: %d floats, size=%d", name, len(buf.Data), buf.Size)
	// 			}
	// 		}
	// 	}
	// }

	var vertexCount int
	indexCount := shader.GetIndexCount()
	if shaderDebug {
		// log.Printf("[drawShader] indexCount=%d", indexCount)
	}

	// Get attribute buffers at function scope so they're accessible in command loop
	attrBuffers := shader.GetAttributeBuffers()

	// Get attribute divisors for instanced rendering (at function scope for command loop)
	attribDivisors := shader.GetAttribDivisors()

	if useSeparateBuffers && !hasRenderCmds {
		// Phase 1 attribute setup: create VBOs, upload data, enable attribs.
		// ONLY runs when there are NO render commands — with render commands,
		// the command loop handles all of this per-draw from snapshots.
		positionBuf := attrBuffers["position"]
		expectedVertexCount := 0
		if positionBuf != nil && positionBuf.Size > 0 && len(positionBuf.Data) > 0 {
			expectedVertexCount = len(positionBuf.Data) / positionBuf.Size
		}

	// Use separate VBOs for each attribute (three.js style)
		if shaderDebug {
			log.Printf("[drawShader] attrs from shader: %d, attrBuffers: %d, divisors: %v", len(attrs), len(attrBuffers), attribDivisors)
			for _, a := range attrs {
				glLoc := p.ctx.GetAttribLocation(prog, a.name)
				log.Printf("[drawShader]   GL location: %s -> %d", a.name, glLoc)
			}
			for _, a := range attrs {
				log.Printf("[drawShader]   attr: name=%s type=%s size=%d", a.name, a.glslType, a.size)
			}
			for name, b := range attrBuffers {
				if b != nil {
					log.Printf("[drawShader]   buf: %s = %d floats, size=%d stride=%d offset=%d", name, len(b.Data), b.Size, b.Stride, b.Offset)
				}
			}
		}
		for _, attr := range attrs {
			// Look for a buffer matching this attribute name
			buf := attrBuffers[attr.name]
			bufName := attr.name
			if buf == nil || len(buf.Data) == 0 {
				// Name didn't match — try layout location fallback.
				// Custom engines (e.g. Charon Jr) use minified attribute names (o, i, t)
				// but buffers are stored by conventional names (position, normal, uv).
				// Map layout location → conventional buffer name.
				if attr.layoutLocation >= 0 {
					var fallbackName string
					switch attr.layoutLocation {
					case 0:
						fallbackName = "position"
					case 1:
						fallbackName = "normal"
					case 2:
						fallbackName = "uv"
					default:
						fallbackName = fmt.Sprintf("attr_%d", attr.layoutLocation)
					}
					if fb := attrBuffers[fallbackName]; fb != nil && len(fb.Data) > 0 {
						buf = fb
						bufName = fallbackName
						if shaderDebug {
							log.Printf("[drawShader] attr %s: name miss, layout(%d) -> buf %s", attr.name, attr.layoutLocation, fallbackName)
						}
					}
				}
				if buf == nil || len(buf.Data) == 0 {
					if shaderDebug {
						log.Printf("[drawShader] Skipping attr %s: no buffer data", attr.name)
					}
					continue
				}
			}

			// Get attribute location - prefer GL query for the compiled program,
			// fall back to layout location or JS-stored location.
			attribLoc := p.ctx.GetAttribLocation(prog, attr.name)
			if attribLoc < 0 && attr.layoutLocation >= 0 {
				attribLoc = Attribute(attr.layoutLocation)
			}
			if attribLoc < 0 {
				storedLoc := shader.GetAttributeLocation(attr.name)
				if storedLoc >= 0 && !mat4Locs[int(storedLoc)] {
					attribLoc = Attribute(storedLoc)
				}
			}
			if attribLoc < 0 {
				continue
			}

			// Check if this is an instanced attribute (divisor > 0)
			isInstanced := attribDivisors[bufName] > 0

			// Validate buffer vertex count matches position (skip mismatched buffers)
			// This prevents using stale buffers from previous geometry.
			// Skip this check for instanced attributes which have instance count, not vertex count.
			if !isInstanced && buf.Size > 0 && expectedVertexCount > 0 && bufName != "position" {
				bufferVertexCount := len(buf.Data) / buf.Size
				if bufferVertexCount != expectedVertexCount {
					continue
				}
			}

			// Create VBO for this attribute if needed
			if buf.VBO == 0 {
				buf.VBO = uint32(p.ctx.CreateBuffer())
				paintVBOCreates++
			}

			// Bind and upload data for this attribute
			p.ctx.BindBuffer(arrayBuffer, Buffer(buf.VBO))
			// Generation-based cache: skip BufferData if this VBO already has this generation
			if shader.VBOUploadedGen == nil {
				shader.VBOUploadedGen = make(map[uint32]uint64)
			}
			if shader.VBOUploadedGen[buf.VBO] != buf.Generation {
				p.ctx.BufferData(arrayBuffer, buf.Data, dynamicDraw)
				shader.VBOUploadedGen[buf.VBO] = buf.Generation
			}

			// Handle mat4 attributes: 4 consecutive vec4 columns sharing one VBO
			if attr.glslType == "mat4" {
				div := attribDivisors[attr.name]
				if shaderDebug {
					log.Printf("[drawShader] mat4 %s: GL loc=%d, div=%d, data=%d floats, stride=%d offset=%d",
						attr.name, attribLoc, div, len(buf.Data), buf.Stride, buf.Offset)
				}
				for col := 0; col < 4; col++ {
					loc := Attribute(int(attribLoc) + col)
					p.ctx.EnableVertexAttribArray(loc)
					enabledAttribs = append(enabledAttribs, loc)
					p.ctx.VertexAttribPointerWithOffset(loc, 4, float, false, buf.Stride, buf.Offset+col*16)
					if div > 0 {
						p.ctx.VertexAttribDivisor(uint32(loc), div)
						if shaderDebug {
							glErr := p.ctx.GetError()
							log.Printf("[drawShader] VertexAttribDivisor(%d, %d) -> glErr=0x%04X", uint32(loc), div, glErr)
						}
					}
				}
			} else {
				// Enable and set up vertex attribute pointer
				// Use stride/offset from the buffer (0,0 = tightly packed)
				p.ctx.EnableVertexAttribArray(attribLoc)
				enabledAttribs = append(enabledAttribs, attribLoc)
				p.ctx.VertexAttribPointerWithOffset(attribLoc, buf.Size, float, false, buf.Stride, buf.Offset)

				// Set divisor for instanced attributes
				if isInstanced {
					if shaderDebug {
						log.Printf("[drawShader] instanced %s: GL loc=%d, div=%d", attr.name, attribLoc, attribDivisors[attr.name])
					}
					p.ctx.VertexAttribDivisor(uint32(attribLoc), attribDivisors[attr.name])
					if shaderDebug {
						glErr := p.ctx.GetError()
						log.Printf("[drawShader] VertexAttribDivisor(%d, %d) -> glErr=0x%04X", uint32(attribLoc), attribDivisors[attr.name], glErr)
					}
				}
			}

			// Calculate vertex count from position attribute (most reliable)
			if attr.name == "position" && buf.Size > 0 {
				vertexCount = len(buf.Data) / buf.Size
			}
		}
	} else if shader.GetVertexCount() > 0 {
		// Legacy interleaved vertex data
		vertices := shader.GetVertices()
		vertexCount = shader.GetVertexCount()

		// Create VBO if needed
		vbo := shader.GetVBO()
		if vbo == 0 {
			vbo = uint32(p.ctx.CreateBuffer())
			paintVBOCreates++
			shader.SetVBO(vbo)
		}
		p.ctx.BindBuffer(arrayBuffer, Buffer(vbo))
		if len(vertices) > 0 {
			p.ctx.BufferData(arrayBuffer, vertices, dynamicDraw)
		}

		// Calculate total stride and set up attributes
		totalStride := 0
		for _, attr := range attrs {
			totalStride += attr.size
		}

		attrOffset := 0
		for _, attr := range attrs {
			// Try stored location first, fall back to GL query
			storedLoc := shader.GetAttributeLocation(attr.name)
			var attribLoc Attribute
			if storedLoc >= 0 {
				attribLoc = Attribute(storedLoc)
			} else {
				attribLoc = p.ctx.GetAttribLocation(prog, attr.name)
			}
			if attribLoc >= 0 {
				p.ctx.EnableVertexAttribArray(attribLoc)
				enabledAttribs = append(enabledAttribs, attribLoc)
				strideBytes := 0
				if len(attrs) > 1 {
					strideBytes = totalStride * 4
				}
				p.ctx.VertexAttribPointerWithOffset(attribLoc, attr.size, float, false, strideBytes, attrOffset*4)
			}
			attrOffset += attr.size
		}
	} else {
		// Default fullscreen quad
		vertices := []float32{
			-1, -1,  // 0: bottom-left
			 1, -1,  // 1: bottom-right
			-1,  1,  // 2: top-left
			 1,  1,  // 3: top-right
		}
		vertexCount = 4

		vbo := shader.GetVBO()
		if vbo == 0 {
			vbo = uint32(p.ctx.CreateBuffer())
			paintVBOCreates++
			shader.SetVBO(vbo)
		}
		p.ctx.BindBuffer(arrayBuffer, Buffer(vbo))
		p.ctx.BufferData(arrayBuffer, vertices, dynamicDraw)

		// Set up 'vert' attribute for fullscreen quad
		storedVertLoc := shader.GetAttributeLocation("vert")
		var vertAttrib Attribute
		if storedVertLoc >= 0 {
			vertAttrib = Attribute(storedVertLoc)
		} else {
			vertAttrib = p.ctx.GetAttribLocation(prog, "vert")
		}
		if vertAttrib >= 0 {
			p.ctx.EnableVertexAttribArray(vertAttrib)
			enabledAttribs = append(enabledAttribs, vertAttrib)
			p.ctx.VertexAttribPointerWithOffset(vertAttrib, 2, float, false, 0, 0)
		}
	}

	// Create and upload IBO if we have indices for indexed drawing
	// Skip when render commands exist — command loop manages its own IBOs per-draw.
	useIndexedDrawing := false
	if indexCount > 0 && useSeparateBuffers && !hasRenderCmds {
		indices := shader.GetIndices()
		if len(indices) > 0 {
			// Create IBO if needed
			ibo := shader.GetIBO()
			if ibo == 0 {
				ibo = uint32(p.ctx.CreateBuffer())
				paintIBOCreates++
				shader.SetIBO(ibo)
			}
			p.ctx.BindBuffer(elementArrayBuffer, Buffer(ibo))
			p.ctx.BufferDataUint16(elementArrayBuffer, indices, dynamicDraw)
			useIndexedDrawing = true
		}
	}

	p.logError()

	if shaderProfile {
		now := time.Now()
		profileAccum.attribNs += now.Sub(phaseStart).Nanoseconds()
		phaseStart = now
	}

	if len(enabledAttribs) == 0 && shaderDebug {
		log.Printf("[drawShader] WARNING: No vertex attributes bound!")
	}

	// Process render command queue (immediate-mode: set uniform → draw → set uniform → draw)
	var cmdLoopElapsed, flushElapsed time.Duration
	var budgetExceeded bool
	renderCommands := shader.GetRenderCommands()
	if len(renderCommands) > 0 {
		// VBO upload cache is persistent across frames: the generation-based check
		// at each upload site handles staleness correctly. If Phase 1 uploaded data
		// with generation X for a VBO, command snapshots with the same generation
		// can safely skip re-upload since the data is identical on the GPU.

		// Drain any pre-existing GL errors before our command loop
		for p.ctx.GetError() != 0 {
		}

		// Set up GL state for 3D rendering (three.js style)
		p.ctx.Enable(depthTest)
		p.ctx.DepthFunc(lequal)
		p.ctx.DepthMask(true)
		p.ctx.Enable(cullFace)
		p.ctx.CullFace(backFace)
		p.ctx.FrontFace(ccw)
		if !isGLESBackend {
			p.ctx.Enable(programPointSize)
		}
		if shaderDebug {
			typeCounts := make(map[string]int)
			for _, cmd := range renderCommands {
				typeCounts[cmd.Type]++
			}
			isNew := "REPLAY"
			if shader.LastGetWasNew {
				isNew = "NEW"
			}
			log.Printf("[drawShader] %s frame: %d commands %v", isNew, len(renderCommands), typeCounts)
		}

		// Reset all attribute divisors before entering the command loop.
		// Phase 1 may have set divisors (e.g. for instanced mesh attributes)
		// on locations that subsequent non-instanced draws will reuse.
		// The command loop sets divisors per-draw as needed and resets after instanced draws.
		for loc := uint32(0); loc < 16; loc++ {
			p.ctx.VertexAttribDivisor(loc, 0)
		}

		// Track which FBO is currently bound so we can translate viewport
		// commands for the default framebuffer (add viewportX/Y offset).
		currentFBOId := 0

		// Execute commands in order (three.js immediate-mode style)
		cmdLoopStart := time.Now()
		var cmdStallCount int
		const paintBudget = 300 * time.Millisecond // break out if command loop exceeds this
		for cmdIdx := 0; cmdIdx < len(renderCommands); cmdIdx++ {
			// Paint budget: if the command loop is taking too long, break out.
			// This clips the tail of slow frames (some draw calls missing for
			// one frame) but prevents the GPU driver from hanging the machine.
			// Check every 32 commands to avoid per-command time.Now() overhead.
			if cmdIdx&31 == 31 {
				if time.Since(cmdLoopStart) > paintBudget {
					log.Printf("[PAINT_BUDGET] frame#%d breaking at cmd %d/%d after %s",
						paintCounter, cmdIdx, len(renderCommands), time.Since(cmdLoopStart))
					budgetExceeded = true
					break
				}
			}
			cmdStart := time.Now()
			cmd := renderCommands[cmdIdx]
			_ = cmdIdx
			switch cmd.Type {
			case "uniform":
				var uniformStart time.Time
				if shaderProfile {
					uniformStart = time.Now()
				}
				setUniform(cmd.Name, cmd.Value)
				if shaderProfile {
					profileAccum.uniformNs += time.Since(uniformStart).Nanoseconds()
					profileAccum.uniformCount++
				}
			case "drawElements":
				if shaderProfile {
					profileAccum.drawCount++
				}
				if params, ok := cmd.Value.(canvas.DrawElementsParams); ok {
					if params.Count > 0 {
						if shaderDebug {
							// Log active program
							log.Printf("[drawShader] drawElements: count=%d mode=%d inst=%d attrs=%d indices=%d activeProg=%d",
								params.Count, params.Mode, params.InstanceCount, len(params.AttributeBuffers), len(params.Indices), prog)
							for name, buf := range params.AttributeBuffers {
								if buf != nil {
									log.Printf("[drawShader]   cmdAttr %s: %d floats, size=%d stride=%d offset=%d",
										name, len(buf.Data), buf.Size, buf.Stride, buf.Offset)
								}
							}
						}
						// Use captured geometry from the draw command if available
						cmdAttrBuffers := params.AttributeBuffers
						cmdIndices := params.Indices
						if len(cmdAttrBuffers) > 0 {
							// Re-upload geometry for this specific draw call
							for _, attr := range attrs {
								buf := cmdAttrBuffers[attr.name]
								cmdBufName := attr.name
								if buf == nil || len(buf.Data) == 0 {
									// Layout location fallback (same as initial setup)
									if attr.layoutLocation >= 0 {
										var fallbackName string
										switch attr.layoutLocation {
										case 0:
											fallbackName = "position"
										case 1:
											fallbackName = "normal"
										case 2:
											fallbackName = "uv"
										default:
											fallbackName = fmt.Sprintf("attr_%d", attr.layoutLocation)
										}
										if fb := cmdAttrBuffers[fallbackName]; fb != nil && len(fb.Data) > 0 {
											buf = fb
											cmdBufName = fallbackName
										}
									}
									if buf == nil || len(buf.Data) == 0 {
										continue
									}
								}
								attribLoc := p.ctx.GetAttribLocation(prog, attr.name)
								if attribLoc < 0 && attr.layoutLocation >= 0 {
									attribLoc = Attribute(attr.layoutLocation)
								}
								if attribLoc < 0 {
									storedLoc := shader.GetAttributeLocation(attr.name)
									if storedLoc >= 0 && !mat4Locs[int(storedLoc)] {
										attribLoc = Attribute(storedLoc)
									}
								}
								if attribLoc < 0 {
									continue
								}
								// Create VBO if needed
								if buf.VBO == 0 {
									buf.VBO = uint32(p.ctx.CreateBuffer())
									paintVBOCreates++
									// Write back so future snapshots inherit this VBO.
									// Try live buffer first; fall back to vboCache which
									// survives PruneAttributeBuffers (the prune for the NEXT
									// object may have already deleted this entry).
									if liveBuf := shader.GetAttributeBuffer(cmdBufName); liveBuf != nil {
										liveBuf.VBO = buf.VBO
									}
									shader.CacheVBO(cmdBufName, buf.VBO)
								}
								p.ctx.BindBuffer(arrayBuffer, Buffer(buf.VBO))
								// Generation-based cache: skip BufferData if this VBO already has this generation
								if shader.VBOUploadedGen == nil {
									shader.VBOUploadedGen = make(map[uint32]uint64)
								}
								if shader.VBOUploadedGen[buf.VBO] != buf.Generation {
									p.ctx.BufferData(arrayBuffer, buf.Data, dynamicDraw)
									shader.VBOUploadedGen[buf.VBO] = buf.Generation
									}
								// Handle mat4: 4 consecutive vec4 columns sharing one VBO
								if attr.glslType == "mat4" {
									div := attribDivisors[cmdBufName]
									for col := 0; col < 4; col++ {
										loc := Attribute(int(attribLoc) + col)
										p.ctx.EnableVertexAttribArray(loc)
										p.ctx.VertexAttribPointerWithOffset(loc, 4, float, false, buf.Stride, buf.Offset+col*16)
										if div > 0 {
											p.ctx.VertexAttribDivisor(uint32(loc), div)
										}
									}
								} else {
									p.ctx.EnableVertexAttribArray(attribLoc)
									p.ctx.VertexAttribPointerWithOffset(attribLoc, buf.Size, float, false, buf.Stride, buf.Offset)
									// Set divisor for instanced attributes
									if div := attribDivisors[cmdBufName]; div > 0 {
										p.ctx.VertexAttribDivisor(uint32(attribLoc), div)
									}
								}
							}
						}
						if len(cmdIndices) > 0 {
							// Re-upload index buffer for this draw call
							ibo := shader.GetIBO()
							if ibo == 0 {
								ibo = uint32(p.ctx.CreateBuffer())
								paintIBOCreates++
								shader.SetIBO(ibo)
							}
							p.ctx.BindBuffer(elementArrayBuffer, Buffer(ibo))
							p.ctx.BufferDataUint16(elementArrayBuffer, cmdIndices, dynamicDraw)
							// Clear any stale GL errors before draw
							if shaderDebug {
								for p.ctx.GetError() != 0 {}
								// Log vertex data and indices
								if posBuf := cmdAttrBuffers["position"]; posBuf != nil && len(posBuf.Data) >= 9 {
									log.Printf("[drawShader] pos[0:9]: %.3f %.3f %.3f | %.3f %.3f %.3f | %.3f %.3f %.3f VBO=%d IBO=%d",
										posBuf.Data[0], posBuf.Data[1], posBuf.Data[2],
										posBuf.Data[3], posBuf.Data[4], posBuf.Data[5],
										posBuf.Data[6], posBuf.Data[7], posBuf.Data[8],
										posBuf.VBO, ibo)
								}
								if len(cmdIndices) >= 6 {
									log.Printf("[drawShader] idx[0:6]: %d %d %d %d %d %d", cmdIndices[0], cmdIndices[1], cmdIndices[2], cmdIndices[3], cmdIndices[4], cmdIndices[5])
								}
								}
							// Validate before issuing draw call (prevents GPU hang from buffer overread)
							vtxCount := getMaxVertexCount(cmdAttrBuffers, attribDivisors)
							if glValidate && !validateDrawElements(cmdIndices, params.Count, params.Offset, vtxCount, params.InstanceCount, cmdIdx, prog) {
								// Skip this draw — index bounds validation failed
							} else {
								// Buffer access validation (stride/offset check)
								maxVtxIdx := getMaxIndex(cmdIndices, params.Count, params.Offset)
								bufAccessOK := !glDiag || validateBufferAccessForDraw(cmdAttrBuffers, attribDivisors, maxVtxIdx, params.InstanceCount, cmdIdx, prog)
								if glDiag {
									logDrawCall("drawElements(cmd)", cmdIdx, prog, params.Mode, 0, params.Count, params.Offset, params.InstanceCount, cmdAttrBuffers, attribDivisors, cmdIndices)
									if !bufAccessOK {
										log.Printf("[GL_DIAG] ^^^ BUFFER ACCESS VIOLATION — %s draw", map[bool]string{true: "SKIPPING", false: "EXECUTING ANYWAY"}[safeGL])
									}
								}
								if !bufAccessOK && safeGL {
									// Skip — buffer access violation
								} else if params.InstanceCount > 0 {
									p.ctx.DrawElementsInstanced(params.Mode, int32(params.Count), unsignedShort, params.Offset, int32(params.InstanceCount))
									for loc := uint32(0); loc < 16; loc++ {
										p.ctx.VertexAttribDivisor(loc, 0)
									}
								} else {
									p.ctx.DrawElements(params.Mode, int32(params.Count), unsignedShort, params.Offset)
								}
							}
						} else if useIndexedDrawing {
							// Fall back to pre-uploaded indices.
							// Use cmdAttrBuffers for validation when available (attributes were
							// re-uploaded from snapshot), otherwise fall back to Phase 1 attrBuffers.
							preIndices := shader.GetIndices()
							validationBufs := attrBuffers
							if len(cmdAttrBuffers) > 0 {
								validationBufs = cmdAttrBuffers
							}
							vtxCount2 := getMaxVertexCount(validationBufs, attribDivisors)
							if glValidate && !validateDrawElements(preIndices, params.Count, params.Offset, vtxCount2, params.InstanceCount, cmdIdx, prog) {
								// Skip this draw — index bounds validation failed
							} else {
								maxVtxIdx2 := getMaxIndex(preIndices, params.Count, params.Offset)
								bufAccessOK2 := !glDiag || validateBufferAccessForDraw(validationBufs, attribDivisors, maxVtxIdx2, params.InstanceCount, cmdIdx, prog)
								if glDiag {
									logDrawCall("drawElements(pre)", cmdIdx, prog, params.Mode, 0, params.Count, params.Offset, params.InstanceCount, validationBufs, attribDivisors, preIndices)
									if !bufAccessOK2 {
										log.Printf("[GL_DIAG] ^^^ BUFFER ACCESS VIOLATION — %s draw", map[bool]string{true: "SKIPPING", false: "EXECUTING ANYWAY"}[safeGL])
									}
								}
								if !bufAccessOK2 && safeGL {
									// Skip — buffer access violation
								} else if params.InstanceCount > 0 {
									p.ctx.DrawElementsInstanced(params.Mode, int32(params.Count), unsignedShort, params.Offset, int32(params.InstanceCount))
									for loc := uint32(0); loc < 16; loc++ {
										p.ctx.VertexAttribDivisor(loc, 0)
									}
								} else {
									p.ctx.DrawElements(params.Mode, int32(params.Count), unsignedShort, params.Offset)
								}
							}
						}
					}
				}
			case "drawArrays":
				if shaderProfile {
					profileAccum.drawCount++
				}
				if params, ok := cmd.Value.(canvas.DrawArraysParams); ok {
					if params.Count > 0 {
						// Use captured geometry from the draw command if available
						cmdAttrBuffers := params.AttributeBuffers
						if len(cmdAttrBuffers) > 0 {
							// Re-upload geometry for this specific draw call
							for _, attr := range attrs {
								buf := cmdAttrBuffers[attr.name]
								daBufName := attr.name
								if buf == nil || len(buf.Data) == 0 {
									// Layout location fallback (same as initial setup)
									if attr.layoutLocation >= 0 {
										var fallbackName string
										switch attr.layoutLocation {
										case 0:
											fallbackName = "position"
										case 1:
											fallbackName = "normal"
										case 2:
											fallbackName = "uv"
										default:
											fallbackName = fmt.Sprintf("attr_%d", attr.layoutLocation)
										}
										if fb := cmdAttrBuffers[fallbackName]; fb != nil && len(fb.Data) > 0 {
											buf = fb
											daBufName = fallbackName
										}
									}
									if buf == nil || len(buf.Data) == 0 {
										continue
									}
								}
								attribLoc := p.ctx.GetAttribLocation(prog, attr.name)
								if attribLoc < 0 && attr.layoutLocation >= 0 {
									attribLoc = Attribute(attr.layoutLocation)
								}
								if attribLoc < 0 {
									storedLoc := shader.GetAttributeLocation(attr.name)
									if storedLoc >= 0 && !mat4Locs[int(storedLoc)] {
										attribLoc = Attribute(storedLoc)
									}
								}
								if attribLoc < 0 {
									continue
								}
								// Create VBO if needed
								if buf.VBO == 0 {
									buf.VBO = uint32(p.ctx.CreateBuffer())
									paintVBOCreates++
									// Write back so future snapshots inherit this VBO.
									if liveBuf := shader.GetAttributeBuffer(daBufName); liveBuf != nil {
										liveBuf.VBO = buf.VBO
									}
									shader.CacheVBO(daBufName, buf.VBO)
								}
								p.ctx.BindBuffer(arrayBuffer, Buffer(buf.VBO))
								// Generation-based cache: skip BufferData if this VBO already has this generation
								if shader.VBOUploadedGen == nil {
									shader.VBOUploadedGen = make(map[uint32]uint64)
								}
								if shader.VBOUploadedGen[buf.VBO] != buf.Generation {
									p.ctx.BufferData(arrayBuffer, buf.Data, dynamicDraw)
									shader.VBOUploadedGen[buf.VBO] = buf.Generation
									}
								// Handle mat4: 4 consecutive vec4 columns sharing one VBO
								if attr.glslType == "mat4" {
									div := attribDivisors[daBufName]
									for col := 0; col < 4; col++ {
										loc := Attribute(int(attribLoc) + col)
										p.ctx.EnableVertexAttribArray(loc)
										p.ctx.VertexAttribPointerWithOffset(loc, 4, float, false, buf.Stride, buf.Offset+col*16)
										if div > 0 {
											p.ctx.VertexAttribDivisor(uint32(loc), div)
										}
									}
								} else {
									p.ctx.EnableVertexAttribArray(attribLoc)
									p.ctx.VertexAttribPointerWithOffset(attribLoc, buf.Size, float, false, buf.Stride, buf.Offset)
									// Set divisor for instanced attributes
									if div := attribDivisors[daBufName]; div > 0 {
										p.ctx.VertexAttribDivisor(uint32(attribLoc), div)
									}
								}
							}
						}
						// Validate before issuing draw call
						daVtxCount := getMaxVertexCount(cmdAttrBuffers, attribDivisors)
						if glValidate && !validateDrawArrays(params.First, params.Count, daVtxCount, params.InstanceCount, cmdIdx, prog) {
							// Skip this draw — validation failed
						} else {
							maxVtxIdx := params.First + params.Count - 1
							if maxVtxIdx < 0 {
								maxVtxIdx = 0
							}
							bufAccessOK := !glDiag || validateBufferAccessForDraw(cmdAttrBuffers, attribDivisors, maxVtxIdx, params.InstanceCount, cmdIdx, prog)
							if glDiag {
								logDrawCall("drawArrays", cmdIdx, prog, params.Mode, params.First, params.Count, 0, params.InstanceCount, cmdAttrBuffers, attribDivisors, nil)
								if !bufAccessOK {
									log.Printf("[GL_DIAG] ^^^ BUFFER ACCESS VIOLATION — %s draw", map[bool]string{true: "SKIPPING", false: "EXECUTING ANYWAY"}[safeGL])
								}
							}
							if !bufAccessOK && safeGL {
								// Skip — buffer access violation
							} else if params.InstanceCount > 0 {
								p.ctx.DrawArraysInstanced(params.Mode, int32(params.First), int32(params.Count), int32(params.InstanceCount))
								for loc := uint32(0); loc < 16; loc++ {
									p.ctx.VertexAttribDivisor(loc, 0)
								}
							} else {
								p.ctx.DrawArrays(params.Mode, params.First, params.Count)
							}
						}
					}
				}
			case "clearColor":
				if cc, ok := cmd.Value.([4]float32); ok {
					p.ctx.ClearColor(cc[0], cc[1], cc[2], cc[3])
				}
			case "clear":
				if mask, ok := cmd.Value.(uint32); ok {
					if shaderDebug {
						log.Printf("[drawShader] clear: mask=0x%x (color=%v depth=%v stencil=%v)",
							mask, mask&0x4000 != 0, mask&0x100 != 0, mask&0x400 != 0)
					}
					// Re-apply stored clear color before clearing
					// Fyne's painter.Clear() resets glClearColor to the theme bg each frame,
					// but Three.js only sends clearColor when the value changes (WebGLState caching)
					cc := shader.GetClearColor()
					p.ctx.ClearColor(cc[0], cc[1], cc[2], cc[3])
					// Use scissor test to limit clear to this shader's viewport region.
					// glClear ignores glViewport and clears the entire framebuffer,
					// which would erase other shaders' content in multi-canvas layouts.
					p.ctx.Enable(scissorTest)
					p.ctx.Scissor(int32(viewportX), int32(viewportY), int32(viewportW), int32(viewportH))
					p.ctx.Clear(mask)
					p.ctx.Disable(scissorTest)
				} else {
					// Legacy path (QueueClear without mask)
					cc := shader.GetClearColor()
					p.ctx.ClearColor(cc[0], cc[1], cc[2], cc[3])
					p.ctx.Enable(scissorTest)
					p.ctx.Scissor(int32(viewportX), int32(viewportY), int32(viewportW), int32(viewportH))
					p.ctx.Clear(colorBufferBit | depthBufferBit)
					p.ctx.Disable(scissorTest)
				}
			case "enable":
				if cap, ok := cmd.Value.(uint32); ok {
					if shaderDebug {
						log.Printf("[drawShader] enable: cap=0x%x", cap)
					}
					p.ctx.Enable(cap)
				}
			case "disable":
				if cap, ok := cmd.Value.(uint32); ok {
					if shaderDebug {
						log.Printf("[drawShader] disable: cap=0x%x", cap)
					}
					p.ctx.Disable(cap)
				}
			case "cullFace":
				if mode, ok := cmd.Value.(uint32); ok {
					p.ctx.CullFace(mode)
				}
			case "blendFunc":
				if factors, ok := cmd.Value.([2]uint32); ok {
					p.ctx.BlendFunc(factors[0], factors[1])
				}
			case "useProgram":
				if params, ok := cmd.Value.(canvas.UseProgramParams); ok {
					cacheKey := shaderCacheKey(params.VertexSource, params.FragmentSource)
					cachedProg, found := shader.GetProgramCache()[cacheKey]
					if shaderDebug {
						log.Printf("[drawShader] useProgram: cached=%v prog=%d vertLen=%d fragLen=%d",
							found, cachedProg, len(params.VertexSource), len(params.FragmentSource))
					}
					if found && cachedProg != 0 {
						prog = Program(cachedProg)
					} else {
						newProg := p.compileShaderProgram(params.VertexSource, params.FragmentSource)
						if newProg != 0 {
							shader.SetProgramCache(cacheKey, uint32(newProg))
							prog = newProg
						}
					}
					p.ctx.UseProgram(prog)
					attrs = parseVertexAttributes(params.VertexSource)
					for k := range shader.UniformLocs() {
						delete(shader.UniformLocs(), k)
					}
					// Rebuild mat4Locs for the new program — stale entries from the
					// previous program can cause attribute location collisions.
					for k := range mat4Locs {
						delete(mat4Locs, k)
					}
					for _, a := range attrs {
						if a.glslType == "mat4" {
							glLoc := p.ctx.GetAttribLocation(prog, a.name)
							if glLoc >= 0 {
								for col := 0; col < 4; col++ {
									mat4Locs[int(glLoc)+col] = true
								}
							}
						}
					}
					// Reset divisors — the previous program may have set instanced
					// divisors on locations that the new program uses differently.
					for loc := uint32(0); loc < 16; loc++ {
						p.ctx.VertexAttribDivisor(loc, 0)
					}
				}

			// ═══ FBO render commands ═══
			case "createFramebuffer":
				jsId, _ := strconv.Atoi(cmd.Name)
				fb := p.ctx.CreateFramebuffer()
				paintFBOCreates++
				shader.SetFBOCache(jsId, uint32(fb))
				if shaderDebug {
					log.Printf("[drawShader] createFramebuffer: js=%d -> gl=%d", jsId, fb)
				}

			case "deleteFramebuffer":
				jsId, _ := strconv.Atoi(cmd.Name)
				glId := shader.GetFBOCache(jsId)
				if glId != 0 {
					p.ctx.DeleteFramebuffer(Framebuffer(glId))
					shader.DeleteFBOCache(jsId)
				}

			case "bindFramebuffer":
				if params, ok := cmd.Value.(canvas.BindFramebufferParams); ok {
					currentFBOId = params.FramebufferId
					if params.FramebufferId == 0 {
						p.ctx.BindFramebuffer(params.Target, Framebuffer(0))
					} else {
						glId := shader.GetFBOCache(params.FramebufferId)
						p.ctx.BindFramebuffer(params.Target, Framebuffer(glId))
						// Check FBO completeness — drawing to an incomplete FBO is
						// undefined behavior that can hang the GPU on some drivers.
						if glValidate {
							status := p.ctx.CheckFramebufferStatus(params.Target)
							if status != 0x8CD5 { // GL_FRAMEBUFFER_COMPLETE
								log.Printf("[GL_VALIDATE] WARNING: FBO %d (gl=%d) incomplete, status=0x%x — draws to this FBO will be skipped",
									params.FramebufferId, glId, status)
							}
						}
					}
				}

			case "framebufferTexture2D":
				if params, ok := cmd.Value.(canvas.FramebufferTexture2DParams); ok {
					// Look up GL texture ID - check GPU tex cache first, then image-based textures
					glTexId := shader.GetGPUTexCache(params.TextureId)
					if glTexId == 0 {
						// Try the image-based texture cache
						for texName := range shader.GetTextures() {
							if cached, ok := shader.GetTextureCache()[texName]; ok {
								// This is a heuristic - we match by checking if the texture ID maps
								// In practice, shadow map textures go through gpuTexCache
								_ = cached
							}
						}
					}
					if glTexId != 0 {
						p.ctx.FramebufferTexture2D(params.Target, params.Attachment, params.Textarget, Texture(glTexId), params.Level)
					} else if shaderDebug {
						log.Printf("[drawShader] framebufferTexture2D: texture %d not found in cache", params.TextureId)
					}
				}

			case "createRenderbuffer":
				jsId, _ := strconv.Atoi(cmd.Name)
				rb := p.ctx.CreateRenderbuffer()
				paintRBOCreates++
				shader.SetRBOCache(jsId, uint32(rb))
				if shaderDebug {
					log.Printf("[drawShader] createRenderbuffer: js=%d -> gl=%d", jsId, rb)
				}

			case "deleteRenderbuffer":
				jsId, _ := strconv.Atoi(cmd.Name)
				glId := shader.GetRBOCache(jsId)
				if glId != 0 {
					p.ctx.DeleteRenderbuffer(Renderbuffer(glId))
					shader.DeleteRBOCache(jsId)
				}

			case "bindRenderbuffer":
				if params, ok := cmd.Value.(canvas.BindFramebufferParams); ok {
					if params.FramebufferId == 0 {
						p.ctx.BindRenderbuffer(params.Target, Renderbuffer(0))
					} else {
						glId := shader.GetRBOCache(params.FramebufferId)
						p.ctx.BindRenderbuffer(params.Target, Renderbuffer(glId))
					}
				}

			case "renderbufferStorage":
				if params, ok := cmd.Value.(canvas.RenderbufferStorageParams); ok {
					p.ctx.RenderbufferStorage(params.Target, params.Internalformat, params.Width, params.Height)
				}

			case "framebufferRenderbuffer":
				if params, ok := cmd.Value.(canvas.FramebufferRenderbufferParams); ok {
					glId := shader.GetRBOCache(params.RenderbufferId)
					p.ctx.FramebufferRenderbuffer(params.Target, params.Attachment, params.RenderbufferTarget, Renderbuffer(glId))
				}

			case "viewport":
				if params, ok := cmd.Value.(canvas.ViewportParams); ok {
					if currentFBOId == 0 {
						// Default framebuffer: translate to widget position
						if shaderDebug {
							log.Printf("[drawShader] viewport cmd: (%d,%d,%d,%d) + offset(%d,%d) => (%d,%d,%d,%d) fbo=%d",
								params.X, params.Y, params.Width, params.Height,
								viewportX, viewportY,
								params.X+viewportX, params.Y+viewportY, params.Width, params.Height,
								currentFBOId)
						}
						p.ctx.Viewport(params.X+viewportX, params.Y+viewportY, params.Width, params.Height)
					} else {
						// FBO rendering: use coordinates as-is
						p.ctx.Viewport(params.X, params.Y, params.Width, params.Height)
					}
				}

			case "colorMask":
				if params, ok := cmd.Value.(canvas.ColorMaskParams); ok {
					p.ctx.ColorMask(params.R, params.G, params.B, params.A)
				}

			case "clearDepth":
				if params, ok := cmd.Value.(canvas.ClearDepthParams); ok {
					if shaderDebug {
						log.Printf("[drawShader] clearDepth: %.4f", params.Depth)
					}
					p.ctx.ClearDepthf(params.Depth)
				}

			case "drawBuffers":
				if params, ok := cmd.Value.(canvas.DrawBuffersParams); ok {
					p.ctx.DrawBuffers(params.Buffers)
				}

			case "texImage2D_gpu":
				// GPU-only texture allocation (e.g., shadow maps with null pixels, DataTexture storage)
				// IMPORTANT: Use scratch texture unit 15 to avoid corrupting sampler bindings
				// set up in Phase 2.1. Without this, BindTexture here overwrites whatever
				// texture unit is active, making Phase 2.1 textures invisible.
				jsTexId, _ := strconv.Atoi(cmd.Name)
				if params, ok := cmd.Value.(canvas.TexImage2DGPUParams); ok {
					// Cubemap face targets: TEXTURE_CUBE_MAP_POSITIVE_X (0x8515) through NEGATIVE_Z (0x851A)
					isCubeFace := params.Target >= 0x8515 && params.Target <= 0x851A
					if isCubeFace {
						// Cubemap face upload: create/reuse ONE GL cubemap texture and upload each face
						faceIndex := int(params.Target - 0x8515)
						p.ctx.ActiveTexture(texture0 + 15) // scratch unit
						existingGlId := shader.GetGPUTexCache(jsTexId)
						if existingGlId == 0 {
							// First face — create the cubemap texture
							tex := p.ctx.CreateTexture()
							paintTexCreates++
							existingGlId = uint32(tex)
							p.ctx.BindTexture(textureCubeMap, Texture(existingGlId))
							p.ctx.TexParameteri(textureCubeMap, textureMinFilter, linear)
							p.ctx.TexParameteri(textureCubeMap, textureMagFilter, linear)
							p.ctx.TexParameteri(textureCubeMap, textureWrapS, clampToEdge)
							p.ctx.TexParameteri(textureCubeMap, textureWrapT, clampToEdge)
							shader.SetGPUTexCache(jsTexId, existingGlId)
							if shaderDebug {
								log.Printf("[drawShader] texImage2D_gpu: CUBEMAP created js=%d -> gl=%d", jsTexId, existingGlId)
							}
						} else {
							p.ctx.BindTexture(textureCubeMap, Texture(existingGlId))
						}
						// Upload face pixel data from CPU image store (compound key: jsTexId*10 + faceIndex)
						faceKey := jsTexId*10 + faceIndex
						if cpuImg := shader.GetCPUTexImage(faceKey); cpuImg != nil {
							rgba := p.textureToRGBA(cpuImg)
							if rgba != nil && len(rgba.Pix) > 0 {
								w := rgba.Rect.Size().X
								h := rgba.Rect.Size().Y
								p.ctx.TexImage2D(params.Target, 0, w, h, colorFormatRGBA, unsignedByte, rgba.Pix)
								// Release CPU-side face data — it's on the GPU now
								shader.ReleaseCPUTexImage(faceKey)
								if shaderDebug {
									log.Printf("[drawShader] texImage2D_gpu: CUBEMAP face %d (%dx%d) uploaded for js=%d", faceIndex, w, h, jsTexId)
								}
							}
						} else if shaderDebug {
							log.Printf("[drawShader] texImage2D_gpu: CUBEMAP face %d missing CPU image (key=%d)", faceIndex, faceKey)
						}
						p.ctx.BindTexture(textureCubeMap, Texture(0))
						p.ctx.ActiveTexture(texture0)
					} else {
						// Standard 2D texture allocation
						// Skip if Phase 2.1 already created and uploaded a texture for this JS ID.
						if existingGlId := shader.GetGPUTexCache(jsTexId); existingGlId != 0 {
							if shaderDebug {
								log.Printf("[drawShader] texImage2D_gpu: SKIPPED js=%d (already have gl=%d)", jsTexId, existingGlId)
							}
						} else {
							p.ctx.ActiveTexture(texture0 + 15) // scratch unit
							tex := p.ctx.CreateTexture()
							paintTexCreates++
							p.ctx.BindTexture(params.Target, tex)
							p.ctx.TexParameteri(params.Target, textureMinFilter, nearest)
							p.ctx.TexParameteri(params.Target, textureMagFilter, nearest)
							p.ctx.TexParameteri(params.Target, textureWrapS, clampToEdge)
							p.ctx.TexParameteri(params.Target, textureWrapT, clampToEdge)
							p.ctx.TexImage2DEmpty(params.Target, params.Level, params.Internalformat, params.Width, params.Height, params.Format, params.Type)
							p.ctx.BindTexture(params.Target, Texture(0)) // unbind from scratch unit
							p.ctx.ActiveTexture(texture0) // restore to unit 0
							shader.SetGPUTexCache(jsTexId, uint32(tex))
							if shaderDebug {
								log.Printf("[drawShader] texImage2D_gpu: js=%d -> gl=%d (%dx%d, fmt=0x%x)", jsTexId, tex, params.Width, params.Height, params.Format)
							}
						}
					}
				}

			case "texStorage3D":
				jsTexId, _ := strconv.Atoi(cmd.Name)
				if params, ok := cmd.Value.(canvas.TexStorage3DGPUParams); ok {
					if existingGlId := shader.GetGPUTexCache(jsTexId); existingGlId != 0 {
						if shaderDebug {
							log.Printf("[drawShader] texStorage3D: SKIPPED js=%d (already have gl=%d)", jsTexId, existingGlId)
						}
					} else {
						p.ctx.ActiveTexture(texture0 + 15)
						tex := p.ctx.CreateTexture()
						paintTexCreates++
						p.ctx.BindTexture(params.Target, tex)
						p.ctx.TexParameteri(params.Target, textureMinFilter, nearest)
						p.ctx.TexParameteri(params.Target, textureMagFilter, nearest)
						p.ctx.TexParameteri(params.Target, textureWrapS, clampToEdge)
						p.ctx.TexParameteri(params.Target, textureWrapT, clampToEdge)
						p.ctx.TexStorage3D(params.Target, int32(params.Levels), params.Internalformat, int32(params.Width), int32(params.Height), int32(params.Depth))
						p.ctx.BindTexture(params.Target, Texture(0))
						p.ctx.ActiveTexture(texture0)
						shader.SetGPUTexCache(jsTexId, uint32(tex))
						if shaderDebug {
							log.Printf("[drawShader] texStorage3D: js=%d -> gl=%d (%dx%dx%d, fmt=0x%x)", jsTexId, tex, params.Width, params.Height, params.Depth, params.Internalformat)
						}
					}
				}

			case "texSubImage3D":
				jsTexId, _ := strconv.Atoi(cmd.Name)
				if params, ok := cmd.Value.(canvas.TexSubImage3DGPUParams); ok {
					glId := shader.GetGPUTexCache(jsTexId)
					if glId != 0 && len(params.Pixels) > 0 {
						p.ctx.ActiveTexture(texture0 + 15)
						p.ctx.BindTexture(params.Target, Texture(glId))
						p.ctx.TexSubImage3D(params.Target, int32(params.Level), int32(params.Xoffset), int32(params.Yoffset), int32(params.Zoffset), int32(params.Width), int32(params.Height), int32(params.Depth), params.Format, params.Type, params.Pixels)
						p.ctx.BindTexture(params.Target, Texture(0))
						p.ctx.ActiveTexture(texture0)
						if shaderDebug {
							log.Printf("[drawShader] texSubImage3D: js=%d gl=%d layer=%d (%dx%d, %d bytes)", jsTexId, glId, params.Zoffset, params.Width, params.Height, len(params.Pixels))
						}
					}
				}

			case "texParameteri":
				if params, ok := cmd.Value.(canvas.TexParameteriParams); ok {
					p.ctx.TexParameteri(params.Target, params.Pname, params.Param)
				}

			case "activeTexture":
				if texEnum, ok := cmd.Value.(uint32); ok {
					p.ctx.ActiveTexture(texEnum)
				}

			case "bindTexture":
				jsTexId, _ := strconv.Atoi(cmd.Name)
				target, _ := cmd.Value.(uint32)
				if jsTexId == 0 {
					// Unbind
					p.ctx.BindTexture(target, Texture(0))
				} else {
					// Check GPU texture cache first (shadow maps etc.)
					if glId := shader.GetGPUTexCache(jsTexId); glId != 0 {
						p.ctx.BindTexture(target, Texture(glId))
						// On-demand upload: if GPU texture was allocated empty (texStorage2D)
						// and we have CPU image data, upload it now.
						// Skip for cubemaps — faces are uploaded by texImage2D_gpu.
						if target != textureCubeMap && !shader.IsGPUTexUploaded(jsTexId) {
							cpuImg := shader.GetCPUTexImage(jsTexId)
							if cpuImg != nil {
								rgba := p.textureToRGBA(cpuImg)
								if rgba != nil && len(rgba.Pix) > 0 {
									p.ctx.TexParameteri(target, textureMinFilter, linear)
									p.ctx.TexParameteri(target, textureMagFilter, linear)
									texFmt := shader.GetGPUTexFormat(jsTexId)
									w := rgba.Rect.Size().X
									h := rgba.Rect.Size().Y
									if texFmt.Format == 0x1903 { // GL_RED
										// Extract R channel only (1 byte/pixel)
										redData := make([]byte, w*h)
										for i := 0; i < w*h; i++ {
											redData[i] = rgba.Pix[i*4]
										}
										p.ctx.TexImage2D(target, 0, w, h, texFmt.Format, unsignedByte, redData)
									} else {
										p.ctx.TexImage2D(target, 0, w, h, colorFormatRGBA, unsignedByte, rgba.Pix)
									}
									shader.SetGPUTexUploaded(jsTexId)
									if shaderDebug {
										log.Printf("[drawShader] bindTexture: ON-DEMAND uploaded %dx%d to js=%d gl=%d (fmt=0x%x)", w, h, jsTexId, glId, texFmt.Format)
									}
								}
							}
						}
					} else if cpuImg := shader.GetCPUTexImage(jsTexId); cpuImg != nil {
						// GPU texture not yet allocated (bindTexture arrived before
						// texImage2D_gpu in the render command queue).  Proactively
						// create the GL texture and upload the CPU image now.
						// Create + bind on the CURRENT active texture unit (set by
						// the preceding activeTexture command).
						texFmt := shader.GetGPUTexFormat(jsTexId)
						rgba := p.textureToRGBA(cpuImg)
						if rgba != nil && len(rgba.Pix) > 0 {
							tex := p.ctx.CreateTexture()
							paintTexCreates++
							p.ctx.BindTexture(target, tex)
							p.ctx.TexParameteri(target, textureMinFilter, linear)
							p.ctx.TexParameteri(target, textureMagFilter, linear)
							p.ctx.TexParameteri(target, textureWrapS, clampToEdge)
							p.ctx.TexParameteri(target, textureWrapT, clampToEdge)
							w := rgba.Rect.Size().X
							h := rgba.Rect.Size().Y
							if texFmt.Format == 0x1903 { // GL_RED
								redData := make([]byte, w*h)
								for i := 0; i < w*h; i++ {
									redData[i] = rgba.Pix[i*4]
								}
								p.ctx.TexImage2D(target, 0, w, h, texFmt.Format, unsignedByte, redData)
							} else {
								p.ctx.TexImage2D(target, 0, w, h, colorFormatRGBA, unsignedByte, rgba.Pix)
							}
							shader.SetGPUTexCache(jsTexId, uint32(tex))
							shader.SetGPUTexUploaded(jsTexId)
							if shaderDebug {
								log.Printf("[drawShader] bindTexture: PROACTIVE created+uploaded %dx%d for js=%d gl=%d", w, h, jsTexId, tex)
							}
						}
					} else if glId := shader.GetTextureCache()[fmt.Sprintf("tex_%d", jsTexId)]; glId != 0 {
						// Fall back to regular texture cache
						p.ctx.BindTexture(target, Texture(glId))
					}
				}

			case "generateMipmap":
				if target, ok := cmd.Value.(uint32); ok {
					p.ctx.GenerateMipmap(target)
				}

			case "depthFunc":
				if fn, ok := cmd.Value.(uint32); ok {
					if shaderDebug {
						log.Printf("[drawShader] depthFunc: 0x%x (LEQUAL=0x%x LESS=0x201 NEVER=0x200)", fn, lequal)
					}
					p.ctx.DepthFunc(fn)
				}

			case "depthMask":
				if flag, ok := cmd.Value.(bool); ok {
					p.ctx.DepthMask(flag)
				}

			case "frontFace":
				if mode, ok := cmd.Value.(uint32); ok {
					p.ctx.FrontFace(mode)
				}

			case "blendFuncSeparate":
				if factors, ok := cmd.Value.([4]uint32); ok {
					p.ctx.BlendFuncSeparate(factors[0], factors[1], factors[2], factors[3])
				}
			case "blendEquation":
				if mode, ok := cmd.Value.(uint32); ok {
					p.ctx.BlendEquation(mode)
				}
			case "blendEquationSeparate":
				if modes, ok := cmd.Value.([2]uint32); ok {
					p.ctx.BlendEquationSeparate(modes[0], modes[1])
				}
			case "blendColor":
				if c, ok := cmd.Value.([4]float32); ok {
					p.ctx.BlendColor(c[0], c[1], c[2], c[3])
				}
			case "polygonOffset":
				if params, ok := cmd.Value.([2]float32); ok {
					p.ctx.PolygonOffset(params[0], params[1])
				}
			case "lineWidth":
				if w, ok := cmd.Value.(float32); ok {
					p.ctx.LineWidth(w)
				}
			case "scissor":
				if rect, ok := cmd.Value.([4]int32); ok {
					p.ctx.Scissor(rect[0], rect[1], rect[2], rect[3])
				}

			case "vertexAttrib1f", "vertexAttrib1fv":
				if packed, ok := cmd.Value.([]float32); ok && len(packed) >= 2 {
					p.ctx.VertexAttrib1f(uint32(packed[0]), packed[1])
				}
			case "vertexAttrib2f", "vertexAttrib2fv":
				if packed, ok := cmd.Value.([]float32); ok && len(packed) >= 3 {
					p.ctx.VertexAttrib2f(uint32(packed[0]), packed[1], packed[2])
				}
			case "vertexAttrib3f", "vertexAttrib3fv":
				if packed, ok := cmd.Value.([]float32); ok && len(packed) >= 4 {
					p.ctx.VertexAttrib3f(uint32(packed[0]), packed[1], packed[2], packed[3])
				}
			case "vertexAttrib4f", "vertexAttrib4fv":
				if packed, ok := cmd.Value.([]float32); ok && len(packed) >= 5 {
					p.ctx.VertexAttrib4f(uint32(packed[0]), packed[1], packed[2], packed[3], packed[4])
				}
			case "vertexAttribDivisor":
				if packed, ok := cmd.Value.([2]uint32); ok {
					p.ctx.VertexAttribDivisor(packed[0], packed[1])
				}
			case "stencilFunc":
				if params, ok := cmd.Value.([3]uint32); ok {
					p.ctx.StencilFunc(params[0], int32(params[1]), params[2])
				}
			case "stencilOp":
				if params, ok := cmd.Value.([3]uint32); ok {
					p.ctx.StencilOp(params[0], params[1], params[2])
				}
			case "stencilMask":
				if mask, ok := cmd.Value.(uint32); ok {
					p.ctx.StencilMask(mask)
				}
			case "stencilFuncSeparate":
				if params, ok := cmd.Value.([4]uint32); ok {
					p.ctx.StencilFuncSeparate(params[0], params[1], int32(params[2]), params[3])
				}
			case "stencilOpSeparate":
				if params, ok := cmd.Value.([4]uint32); ok {
					p.ctx.StencilOpSeparate(params[0], params[1], params[2], params[3])
				}
			case "stencilMaskSeparate":
				if params, ok := cmd.Value.([2]uint32); ok {
					p.ctx.StencilMaskSeparate(params[0], params[1])
				}
			case "clearStencil":
				if val, ok := cmd.Value.(int32); ok {
					p.ctx.ClearStencil(val)
				}
			}
			// Per-command stall detection: log if any single command takes >100ms
			if cmdElapsed := time.Since(cmdStart); cmdElapsed > 100*time.Millisecond {
				cmdStallCount++
				if cmdStallCount <= 10 {
					log.Printf("[CMD_STALL] frame#%d cmd#%d/%d type=%s name=%q took %s",
						paintCounter, cmdIdx, len(renderCommands), cmd.Type, cmd.Name, cmdElapsed)
					// Dump full geometry stats for draw calls that stall
					switch cmd.Type {
					case "drawElements":
						if params, ok := cmd.Value.(canvas.DrawElementsParams); ok {
							logDrawCall("drawElements(STALL)", cmdIdx, prog, params.Mode, 0, params.Count, params.Offset,
								params.InstanceCount, params.AttributeBuffers, shader.GetAttribDivisors(), params.Indices)
						}
					case "drawArrays":
						if params, ok := cmd.Value.(canvas.DrawArraysParams); ok {
							logDrawCall("drawArrays(STALL)", cmdIdx, prog, params.Mode, params.First, params.Count, 0,
								params.InstanceCount, params.AttributeBuffers, shader.GetAttribDivisors(), nil)
						}
					}
				}
			}
		}
		if cmdStallCount > 10 {
			log.Printf("[CMD_STALL] frame#%d: %d more stalled commands (>100ms each)", paintCounter, cmdStallCount-10)
		}

		// GPU pipeline sync — block until the GPU finishes all queued commands.
		// This was previously Flush (non-blocking), but that allowed the GPU command
		// queue to build up across frames when the CPU issued commands faster than
		// the GPU could execute them, eventually causing a driver stall.
		// glFinish was originally removed (Round 4) because it deadlocked when stale
		// attribute bindings caused GPU out-of-bounds reads. That root cause was fixed
		// in Round 6 (enableVertexAttribArray tracking + PruneAttributeBuffers).
		// With validated draws, glFinish is safe and prevents GPU queue buildup.
		cmdLoopElapsed = time.Since(cmdLoopStart)
		flushStart := time.Now()
		p.ctx.Finish()
		flushElapsed = time.Since(flushStart)

		if shaderProfile {
			now := time.Now()
			profileAccum.cmdLoopNs += now.Sub(phaseStart).Nanoseconds()
			phaseStart = now
		}

		// Restore GL state
		p.ctx.BindFramebuffer(framebuffer, Framebuffer(0)) // Unbind any FBO
		p.ctx.ColorMask(true, true, true, true) // Restore color mask
		p.ctx.Disable(depthTest)
		p.ctx.Disable(cullFace)
		if !isGLESBackend {
			p.ctx.Disable(programPointSize)
		}
	} else if shader.VertexSource == "" {
		// Fallback: single draw (for simple shaders without command queue).
		// Only for built-in vertex shader (2D effects). 3D shaders with custom
		// vertex sources must use the render command queue — drawing a fullscreen
		// quad through a 3D vertex shader produces garbage (white flash).
		fbVtxCount := getMaxVertexCount(attrBuffers, attribDivisors)
		if useIndexedDrawing && indexCount > 0 {
			fbIndices := shader.GetIndices()
			if !glValidate || validateDrawElements(fbIndices, indexCount, 0, fbVtxCount, 0, -1, prog) {
				drawOK := true
				if glDiag {
					maxVtxIdx := getMaxIndex(fbIndices, indexCount, 0)
					bufAccessOK := validateBufferAccessForDraw(attrBuffers, attribDivisors, maxVtxIdx, 0, -1, prog)
					logDrawCall("drawElements(fallback)", -1, prog, triangles, 0, indexCount, 0, 0, attrBuffers, attribDivisors, fbIndices)
					if !bufAccessOK {
						log.Printf("[GL_DIAG] ^^^ BUFFER ACCESS VIOLATION (fallback)")
						if safeGL {
							drawOK = false
						}
					}
				}
				if drawOK {
					p.ctx.DrawElements(triangles, int32(indexCount), unsignedShort, 0)
				}
			}
		} else if vertexCount > 0 {
			if !glValidate || validateDrawArrays(0, vertexCount, fbVtxCount, 0, -1, prog) {
				drawOK := true
				if glDiag {
					bufAccessOK := validateBufferAccessForDraw(attrBuffers, attribDivisors, vertexCount-1, 0, -1, prog)
					logDrawCall("drawArrays(fallback)", -1, prog, triangles, 0, vertexCount, 0, 0, attrBuffers, attribDivisors, nil)
					if !bufAccessOK {
						log.Printf("[GL_DIAG] ^^^ BUFFER ACCESS VIOLATION (fallback)")
						if safeGL {
							drawOK = false
						}
					}
				}
				if drawOK {
					if useSeparateBuffers || shader.GetVertexCount() > 0 {
						p.ctx.DrawArrays(triangles, 0, vertexCount)
					} else {
						p.ctx.DrawArrays(triangleStrip, 0, vertexCount)
					}
				}
			}
		}
	}
	p.logError()

	// Capture-on-next-paint: if a capture was requested, read pixels now
	// (while the back buffer still has our rendered content, before SwapBuffers)
	if capChan := shader.GetCaptureChan(); capChan != nil {
		// Restore full viewport for capture (we want the full window)
		capFrameW := int(p.pixScale * frame.Width)
		capFrameH := int(p.pixScale * frame.Height)
		p.ctx.Viewport(0, 0, capFrameW, capFrameH)
		pixels := make([]uint8, capFrameW*capFrameH*4)
		p.ctx.ReadPixels(0, 0, capFrameW, capFrameH, colorFormatRGBA, unsignedByte, pixels)
		capChan <- canvas.CaptureResult{Pixels: pixels, Width: capFrameW, Height: capFrameH}
	}

	// ReadPixels-on-next-paint: if a readPixels was requested, execute it now
	if req, ch := shader.GetReadPixelsReq(); req != nil && ch != nil {
		pixels := make([]uint8, req.Width*req.Height*4)
		p.ctx.ReadPixels(req.X+viewportX, req.Y+viewportY, req.Width, req.Height, colorFormatRGBA, unsignedByte, pixels)
		ch <- canvas.ReadPixelsResult{Pixels: pixels}
	}

	// Disable enabled vertex attributes
	for _, attrib := range enabledAttribs {
		p.ctx.DisableVertexAttribArray(attrib)
	}

	// Unbind and delete the shader VAO, restore VAO 0 for Fyne's default rendering
	p.ctx.BindVertexArray(0)
	p.ctx.DeleteVertexArray(shaderVAO)

	// Restore original viewport (full frame)
	frameW := int(p.pixScale * frame.Width)
	frameH := int(p.pixScale * frame.Height)
	p.ctx.Viewport(0, 0, frameW, frameH)

	// Restore default program
	p.ctx.UseProgram(p.program.ref)

	// Re-enable default program vertex attribute arrays
	// (shader's DisableVertexAttribArray may have disabled them if indices match)
	p.ctx.EnableVertexAttribArray(p.program.attributes["vert"])
	p.ctx.EnableVertexAttribArray(p.program.attributes["vertTexCoord"])

	if shaderProfile {
		now := time.Now()
		profileAccum.cleanupNs += now.Sub(phaseStart).Nanoseconds()
		profileAccum.totalNs += now.Sub(profileStart).Nanoseconds()
		profileAccum.count++
		if profileAccum.count % 60 == 0 {
			n := int64(profileAccum.count)
			log.Printf("[PROFILE] %d frames: total=%.1fms compile=%.1fms tex=%.1fms attr=%.1fms cmdLoop=%.1fms cleanup=%.1fms | uniforms: %d calls %.1fms | draws: %d",
				profileAccum.count,
				float64(profileAccum.totalNs)/float64(n)/1e6,
				float64(profileAccum.compileNs)/float64(n)/1e6,
				float64(profileAccum.textureNs)/float64(n)/1e6,
				float64(profileAccum.attribNs)/float64(n)/1e6,
				float64(profileAccum.cmdLoopNs)/float64(n)/1e6,
				float64(profileAccum.cleanupNs)/float64(n)/1e6,
				profileAccum.uniformCount/int(n),
				float64(profileAccum.uniformNs)/float64(n)/1e6,
				profileAccum.drawCount/int(n),
			)
		}
	}

	// Log slow paint cycles to diagnose GPU hangs / driver stalls.
	// This runs on the GL thread — any delay here blocks frame delivery.
	paintElapsed := time.Since(paintStart)
	if paintElapsed > 200*time.Millisecond {
		budgetStr := ""
		if budgetExceeded {
			budgetStr = " BUDGET_CLIPPED"
		}
		log.Printf("[PAINT_SLOW] frame#%d took %s — cmdLoop=%s flush=%s cleanup=%s cmds=%d%s",
			paintCounter, paintElapsed, cmdLoopElapsed, flushElapsed,
			paintElapsed-cmdLoopElapsed-flushElapsed, len(renderCommands), budgetStr)
	}

	// GPU resource leak detection: log if any VBOs/textures/FBOs were created after warmup.
	// After the first ~60 frames all resources should be cached. New creates = leak.
	if paintCounter > 60 && (paintVBOCreates > 0 || paintTexCreates > 0 || paintFBOCreates > 0 || paintRBOCreates > 0) {
		log.Printf("[GPU_LEAK] frame#%d: VBO=%d IBO=%d tex=%d FBO=%d RBO=%d — resources created after warmup",
			paintCounter, paintVBOCreates, paintIBOCreates, paintTexCreates, paintFBOCreates, paintRBOCreates)
	}

	// Periodic glGetError check (every ~5s at 30fps) — catch GL errors that precede hangs.
	if paintCounter%150 == 0 {
		glErr := p.ctx.GetError()
		if glErr != 0 {
			log.Printf("[GL_ERR] frame#%d: glGetError()=0x%04X after paint", paintCounter, glErr)
		}
	}

	// SignalPaintDone is handled by defer at function entry

}
