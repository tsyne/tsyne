package gl

import (
	"fmt"
	"log"
	"os"
	"strconv"

	"fyne.io/fyne/v2"
	"fyne.io/fyne/v2/canvas"
)

// shaderDebug controls verbose shader logging (set TSYNE_SHADER_DEBUG=1 to enable)
var shaderDebug = os.Getenv("TSYNE_SHADER_DEBUG") == "1"

// shaderAttribute represents a parsed vertex attribute declaration
type shaderAttribute struct {
	name     string
	glslType string  // vec2, vec3, vec4, float, etc.
	size     int     // number of floats (2 for vec2, 3 for vec3, etc.)
}


// Default vertex shader for fullscreen quad
// Uses 'vert' attribute to match Fyne's existing pattern
const defaultShaderVertexSrc = `
#version 110
attribute vec2 vert;

void main() {
    gl_Position = vec4(vert, 0.0, 1.0);
}
`

// Fragment shader wrapper that adds standard uniforms
// Note: No precision qualifier - that's OpenGL ES syntax, not desktop OpenGL
const shaderFragmentPrefix = `
#version 110
uniform vec2 u_resolution;
uniform float u_time;
varying vec2 v_texCoord;
`

// shaderVBO is a dedicated vertex buffer for shader rendering
var shaderVBO Buffer
var shaderVBOInit bool

// drawShader renders a Shader canvas object
func (p *painter) drawShader(shader *canvas.Shader, pos fyne.Position, frame fyne.Size) {
	if shader.FragmentSource == "" {
		return
	}

	// If shader has a custom vertex source (3D/geometry mode), wait for attribute data
	// This prevents rendering a white rectangle before geometry data arrives
	if shader.VertexSource != "" && !shader.HasAttributeBuffers() && shader.GetVertexCount() == 0 {
		// Custom vertex shader but no geometry data yet - skip rendering
		return
	}

	// Compile shader if needed
	// Check both needsCompile flag AND program cache to detect source changes.
	// glUseProgram uses direct field assignment (no needsCompile) to avoid
	// redundant recompilation in multi-program scenes. We detect the need to
	// compile by checking if the current sources have a cached program.
	vertexSrc := shader.VertexSource
	if vertexSrc == "" {
		vertexSrc = defaultShaderVertexSrc
	}
	cacheKey := vertexSrc + "|||" + shader.FragmentSource
	cachedProg, hasCachedProg := shader.GetProgramCache()[cacheKey]
	needsCompile := shader.NeedsCompile()

	if hasCachedProg && cachedProg != 0 && !needsCompile {
		// Sources match a cached program - use it without recompiling
		if shader.Program() != cachedProg {
			shader.SetProgram(cachedProg)
			// Clear uniform location cache when switching programs
			for k := range shader.UniformLocs() {
				delete(shader.UniformLocs(), k)
			}
		}
	} else if needsCompile || shader.Program() == 0 || !hasCachedProg {
		prog := p.compileShaderProgram(vertexSrc, shader.FragmentSource)
		if prog == 0 {
			return
		}
		shader.SetProgram(uint32(prog))
		// Cache program so subsequent frames and useProgram commands get cache hits
		shader.SetProgramCache(cacheKey, uint32(prog))
		// Clear uniform location cache on recompile
		for k := range shader.UniformLocs() {
			delete(shader.UniformLocs(), k)
		}
	}

	prog := Program(shader.Program())
	if prog == 0 {
		return
	}
	if shaderDebug {
		log.Printf("[drawShader] Using program %d (vertexSrc=%d bytes, fragSrc=%d bytes)", prog, len(vertexSrc), len(shader.FragmentSource))
		if len(shader.FragmentSource) > 10000 {
			os.WriteFile("/tmp/tsyne_frag.glsl", []byte(shader.FragmentSource), 0644)
			os.WriteFile("/tmp/tsyne_vert.glsl", []byte(vertexSrc), 0644)
		}
	}

	// Save current program
	p.ctx.UseProgram(prog)

	// Get or cache uniform locations
	getUniformLoc := func(name string) Uniform {
		if loc, ok := shader.UniformLocs()[name]; ok {
			return Uniform(loc)
		}
		loc := p.ctx.GetUniformLocation(prog, name)
		shader.SetUniformLoc(name, int32(loc))
		return loc
	}

	// Calculate pixel position for uniforms
	// Note: OpenGL has origin at bottom-left, Fyne has origin at top-left
	pixelX := p.pixScale * pos.X
	// For gl_FragCoord, y=0 is at bottom of window
	// Shader bottom edge in Fyne coords: pos.Y + shader.Size().Height
	// In OpenGL coords: frame.Height - (pos.Y + shader.Size().Height)
	pixelY := p.pixScale * (frame.Height - pos.Y - shader.Size().Height)
	pixelWidth := p.pixScale * shader.Size().Width
	pixelHeight := p.pixScale * shader.Size().Height

	// Set standard uniforms
	resLoc := getUniformLoc("u_resolution")
	if resLoc >= 0 {
		p.ctx.Uniform2f(resLoc, pixelWidth, pixelHeight)
	}

	// Set viewport to the shader's area so gl_FragCoord is relative to it
	// This makes the shader's coordinate system match what it expects
	viewportX := int(pixelX)
	viewportY := int(pixelY)
	viewportW := int(pixelWidth)
	viewportH := int(pixelHeight)
	p.ctx.Viewport(viewportX, viewportY, viewportW, viewportH)

	// With viewport set, u_position should be 0,0 since gl_FragCoord is now relative to viewport
	posLoc := getUniformLoc("u_position")
	if posLoc >= 0 {
		p.ctx.Uniform2f(posLoc, 0, 0)
	}

	timeLoc := getUniformLoc("u_time")
	if timeLoc >= 0 {
		// Use the shader's cached time - only updated when Refresh() is called
		// This matches Fyne's texture caching behavior where widgets only update on explicit refresh
		p.ctx.Uniform1f(timeLoc, shader.CachedTime())
	}

	// Helper function to set a uniform value
	setUniform := func(name string, val interface{}) {
		loc := getUniformLoc(name)
		if loc < 0 {
			return
		}
		switch v := val.(type) {
		case float32:
			p.ctx.Uniform1f(loc, v)
		case float64:
			p.ctx.Uniform1f(loc, float32(v))
		case int:
			p.ctx.Uniform1f(loc, float32(v))
		case int8:
			p.ctx.Uniform1f(loc, float32(v))
		case int16:
			p.ctx.Uniform1f(loc, float32(v))
		case int32:
			p.ctx.Uniform1i(loc, v)
		case int64:
			p.ctx.Uniform1f(loc, float32(v))
		case uint8:
			p.ctx.Uniform1f(loc, float32(v))
		case []int32:
			if len(v) >= 1 {
				p.ctx.Uniform1i(loc, v[0])
			}
		case [2]float32:
			p.ctx.Uniform2f(loc, v[0], v[1])
		case [3]float32:
			p.ctx.Uniform3f(loc, v[0], v[1], v[2])
		case [4]float32:
			p.ctx.Uniform4f(loc, v[0], v[1], v[2], v[3])
		case []float32:
			switch len(v) {
			case 1:
				p.ctx.Uniform1f(loc, v[0])
			case 2:
				p.ctx.Uniform2f(loc, v[0], v[1])
			case 3:
				p.ctx.Uniform3f(loc, v[0], v[1], v[2])
			case 4:
				p.ctx.Uniform4f(loc, v[0], v[1], v[2], v[3])
			case 9:
				p.ctx.UniformMatrix3fv(loc, false, v)
			case 16:
				p.ctx.UniformMatrix4fv(loc, false, v)
			}
		case []interface{}:
			floats := make([]float32, len(v))
			for i, x := range v {
				switch f := x.(type) {
				case float64:
					floats[i] = float32(f)
				case float32:
					floats[i] = f
				case int:
					floats[i] = float32(f)
				}
			}
			switch len(floats) {
			case 1:
				p.ctx.Uniform1f(loc, floats[0])
			case 2:
				p.ctx.Uniform2f(loc, floats[0], floats[1])
			case 3:
				p.ctx.Uniform3f(loc, floats[0], floats[1], floats[2])
			case 4:
				p.ctx.Uniform4f(loc, floats[0], floats[1], floats[2], floats[3])
			}
		}
	}

	// Set texture uniforms (Phase 2.1)
	// Upload texture data and bind to sampler uniforms
	textureUnit := 0
	for texName, texData := range shader.GetTextures() {
		if textureUnit >= 16 {
			break
		}

		// Get or create GL texture
		texID, cached := shader.GetTextureCache()[texName]
		if !cached {
			tex := p.ctx.CreateTexture()
			texID = uint32(tex)
			shader.SetTextureCache(texName, texID)
		}

		// Activate texture unit and bind texture
		p.ctx.ActiveTexture(texture0 + uint32(textureUnit))
		p.ctx.BindTexture(texture2D, Texture(texID))

		// Upload texture data if not cached (or always upload for now)
		if !cached {
			// Set texture parameters
			p.ctx.TexParameteri(texture2D, textureMinFilter, linear)
			p.ctx.TexParameteri(texture2D, textureMagFilter, linear)
			p.ctx.TexParameteri(texture2D, textureWrapS, clampToEdge)
			p.ctx.TexParameteri(texture2D, textureWrapT, clampToEdge)

			// Convert texture data to RGBA
			rgba := p.textureToRGBA(texData)
			if rgba != nil && len(rgba.Pix) > 0 {
				// Check if this texture has a JS ID with RED format info
				jsTexId := shader.GetJSTexForSampler(texName)
				texFmt := shader.GetGPUTexFormat(jsTexId)
				w := rgba.Rect.Size().X
				h := rgba.Rect.Size().Y
				if texFmt.Format == 0x1903 { // GL_RED
					redData := make([]byte, w*h)
					for i := 0; i < w*h; i++ {
						redData[i] = rgba.Pix[i*4]
					}
					p.ctx.TexImage2D(texture2D, 0, w, h, texFmt.Format, unsignedByte, redData)
				} else {
					p.ctx.TexImage2D(texture2D, 0, w, h, colorFormatRGBA, unsignedByte, rgba.Pix)
				}
			}
		}

		// Set sampler uniform to texture unit
		shader.SetTextureUnit(texName, textureUnit)
		samplerLoc := getUniformLoc(texName)
		if samplerLoc >= 0 {
			p.ctx.Uniform1i(samplerLoc, int32(textureUnit))
		}

		// If this sampler has a JS texture ID mapping, populate gpuTexCache
		// so that bindTexture render commands can find this uploaded texture
		// (overwriting the empty GPU texture created by texImage2D_gpu/texStorage2D)
		if jsTexId := shader.GetJSTexForSampler(texName); jsTexId != 0 {
			shader.SetGPUTexCache(jsTexId, texID)
		}

		textureUnit++
	}

	// Set cubemap uniforms (Phase 2.2)
	// Full cubemap texture upload and binding implementation
	for cubemapName, faces := range shader.GetCubemaps() {
		if textureUnit >= 16 {
			log.Printf("[drawShader] Warning: too many textures+cubemaps (max 16), skipping %s", cubemapName)
			break
		}

		// Get or create cubemap texture
		cubemapTexID, cached := shader.GetCubemapCache()[cubemapName]
		if !cached {
			// Create new cubemap texture
			tex := p.ctx.CreateTexture()
			cubemapTexID = uint32(tex)
			shader.SetCubemapCache(cubemapName, cubemapTexID)
		}

		// Activate texture unit and bind cubemap
		p.ctx.ActiveTexture(texture0 + uint32(textureUnit))
		p.ctx.BindTexture(textureCube, Texture(cubemapTexID))

		// Upload faces if not cached (or always upload for now - caching optimization later)
		if !cached {
			// Set cubemap texture parameters
			p.ctx.TexParameteri(textureCube, textureMinFilter, linear)
			p.ctx.TexParameteri(textureCube, textureMagFilter, linear)
			p.ctx.TexParameteri(textureCube, textureWrapS, clampToEdge)
			p.ctx.TexParameteri(textureCube, textureWrapT, clampToEdge)

			// Upload each of the 6 faces
			faceTargets := []uint32{
				textureCubePositiveX, // +X (right)
				textureCubeNegativeX, // -X (left)
				textureCubePositiveY, // +Y (up)
				textureCubeNegativeY, // -Y (down)
				textureCubePositiveZ, // +Z (front)
				textureCubeNegativeZ, // -Z (back)
			}

			for i, faceTarget := range faceTargets {
				faceData := faces[i]
				if faceData == nil {
					continue
				}

				// Convert face data to RGBA pixel data
				rgba := p.cubemapFaceToRGBA(faceData)
				if rgba == nil || len(rgba.Pix) == 0 {
					log.Printf("[drawShader] Warning: empty face data for cubemap %s face %d", cubemapName, i)
					continue
				}

				// Upload face to cubemap
				p.ctx.TexImage2D(
					faceTarget,
					0,
					rgba.Rect.Size().X,
					rgba.Rect.Size().Y,
					colorFormatRGBA,
					unsignedByte,
					rgba.Pix,
				)
			}
			p.logError()
		}

		// Set samplerCube uniform to texture unit number
		shader.SetCubemapUnit(cubemapName, textureUnit)
		samplerLoc := getUniformLoc(cubemapName)
		if samplerLoc >= 0 {
			p.ctx.Uniform1i(samplerLoc, int32(textureUnit))
		}
		textureUnit++
	}

	// Parse vertex shader to find attributes dynamically
	// (vertexSrc was already computed above for the compilation check)
	attrs := parseVertexAttributes(vertexSrc)
	// if shaderDebug {
	// 	log.Printf("[drawShader] Parsed %d attributes from vertex shader", len(attrs))
	// 	for _, a := range attrs {
	// 		log.Printf("[drawShader]   parsed attr: %s (%s, size=%d)", a.name, a.glslType, a.size)
	// 	}
	// }

	// Build a set of GL locations reserved by mat4 attributes.
	// mat4 attributes occupy 4 consecutive locations (one per column).
	// We must prevent the stored-location fallback from binding other
	// attributes to locations that overlap with mat4 columns.
	mat4Locs := make(map[int]bool)
	for _, a := range attrs {
		if a.glslType == "mat4" {
			glLoc := p.ctx.GetAttribLocation(prog, a.name)
			if glLoc >= 0 {
				for col := 0; col < 4; col++ {
					mat4Locs[int(glLoc)+col] = true
				}
			}
		}
	}

	// Track enabled attributes for cleanup
	var enabledAttribs []Attribute

	// Create a VAO for this shader draw.
	// On modern GL contexts (3.0+), vertex attribute state (including divisors)
	// is per-VAO. Without a bound VAO, VertexAttribDivisor may not take effect.
	shaderVAO := p.ctx.GenVertexArray()
	p.ctx.BindVertexArray(shaderVAO)

	// Check if we have separate attribute buffers (three.js style)
	useSeparateBuffers := shader.HasAttributeBuffers()
	// if shaderDebug {
	// 	log.Printf("[drawShader] useSeparateBuffers=%v, HasAttributeBuffers=%v", useSeparateBuffers, shader.HasAttributeBuffers())
	// 	if shader.AttributeBuffers != nil {
	// 		log.Printf("[drawShader] AttributeBuffers count: %d", len(shader.AttributeBuffers))
	// 		for name, buf := range shader.AttributeBuffers {
	// 			if buf != nil {
	// 				log.Printf("[drawShader]   attr %s: %d floats, size=%d", name, len(buf.Data), buf.Size)
	// 			}
	// 		}
	// 	}
	// }

	var vertexCount int
	indexCount := shader.GetIndexCount()
	if shaderDebug {
		// log.Printf("[drawShader] indexCount=%d", indexCount)
	}

	// Get attribute buffers at function scope so they're accessible in command loop
	attrBuffers := shader.GetAttributeBuffers()

	// Get attribute divisors for instanced rendering (at function scope for command loop)
	attribDivisors := shader.GetAttribDivisors()

	if useSeparateBuffers {
		// First, determine expected vertex count from position attribute
		positionBuf := attrBuffers["position"]
		expectedVertexCount := 0
		if positionBuf != nil && positionBuf.Size > 0 && len(positionBuf.Data) > 0 {
			expectedVertexCount = len(positionBuf.Data) / positionBuf.Size
		}

	// Use separate VBOs for each attribute (three.js style)
		if shaderDebug {
			log.Printf("[drawShader] attrs from shader: %d, attrBuffers: %d, divisors: %v", len(attrs), len(attrBuffers), attribDivisors)
			for _, a := range attrs {
				glLoc := p.ctx.GetAttribLocation(prog, a.name)
				log.Printf("[drawShader]   GL location: %s -> %d", a.name, glLoc)
			}
			for _, a := range attrs {
				log.Printf("[drawShader]   attr: name=%s type=%s size=%d", a.name, a.glslType, a.size)
			}
			for name, b := range attrBuffers {
				if b != nil {
					log.Printf("[drawShader]   buf: %s = %d floats, size=%d stride=%d offset=%d", name, len(b.Data), b.Size, b.Stride, b.Offset)
				}
			}
		}
		for _, attr := range attrs {
			// Look for a buffer matching this attribute name
			buf := attrBuffers[attr.name]
			if buf == nil || len(buf.Data) == 0 {
				if shaderDebug {
					log.Printf("[drawShader] Skipping attr %s: no buffer data", attr.name)
				}
				continue
			}

			// Get attribute location - prefer GL query for the compiled program,
			// fall back to JS-stored location only if GL returns -1 (optimized out)
			// and the stored location doesn't conflict with a mat4 attribute's columns.
			attribLoc := p.ctx.GetAttribLocation(prog, attr.name)
			if attribLoc < 0 {
				storedLoc := shader.GetAttributeLocation(attr.name)
				if storedLoc >= 0 && !mat4Locs[int(storedLoc)] {
					attribLoc = Attribute(storedLoc)
				}
			}
			if attribLoc < 0 {
				continue
			}

			// Check if this is an instanced attribute (divisor > 0)
			isInstanced := attribDivisors[attr.name] > 0

			// Validate buffer vertex count matches position (skip mismatched buffers)
			// This prevents using stale buffers from previous geometry.
			// Skip this check for instanced attributes which have instance count, not vertex count.
			if !isInstanced && buf.Size > 0 && expectedVertexCount > 0 && attr.name != "position" {
				bufferVertexCount := len(buf.Data) / buf.Size
				if bufferVertexCount != expectedVertexCount {
					continue
				}
			}

			// Create VBO for this attribute if needed
			if buf.VBO == 0 {
				buf.VBO = uint32(p.ctx.CreateBuffer())
			}

			// Bind and upload data for this attribute
			p.ctx.BindBuffer(arrayBuffer, Buffer(buf.VBO))
			p.ctx.BufferData(arrayBuffer, buf.Data, dynamicDraw)

			// Handle mat4 attributes: 4 consecutive vec4 columns sharing one VBO
			if attr.glslType == "mat4" {
				div := attribDivisors[attr.name]
				if shaderDebug {
					log.Printf("[drawShader] mat4 %s: GL loc=%d, div=%d, data=%d floats, stride=%d offset=%d",
						attr.name, attribLoc, div, len(buf.Data), buf.Stride, buf.Offset)
				}
				for col := 0; col < 4; col++ {
					loc := Attribute(int(attribLoc) + col)
					p.ctx.EnableVertexAttribArray(loc)
					enabledAttribs = append(enabledAttribs, loc)
					p.ctx.VertexAttribPointerWithOffset(loc, 4, float, false, buf.Stride, buf.Offset+col*16)
					if div > 0 {
						p.ctx.VertexAttribDivisor(uint32(loc), div)
						if shaderDebug {
							glErr := p.ctx.GetError()
							log.Printf("[drawShader] VertexAttribDivisor(%d, %d) -> glErr=0x%04X", uint32(loc), div, glErr)
						}
					}
				}
			} else {
				// Enable and set up vertex attribute pointer
				// Use stride/offset from the buffer (0,0 = tightly packed)
				p.ctx.EnableVertexAttribArray(attribLoc)
				enabledAttribs = append(enabledAttribs, attribLoc)
				p.ctx.VertexAttribPointerWithOffset(attribLoc, buf.Size, float, false, buf.Stride, buf.Offset)

				// Set divisor for instanced attributes
				if isInstanced {
					if shaderDebug {
						log.Printf("[drawShader] instanced %s: GL loc=%d, div=%d", attr.name, attribLoc, attribDivisors[attr.name])
					}
					p.ctx.VertexAttribDivisor(uint32(attribLoc), attribDivisors[attr.name])
					if shaderDebug {
						glErr := p.ctx.GetError()
						log.Printf("[drawShader] VertexAttribDivisor(%d, %d) -> glErr=0x%04X", uint32(attribLoc), attribDivisors[attr.name], glErr)
					}
				}
			}

			// Calculate vertex count from position attribute (most reliable)
			if attr.name == "position" && buf.Size > 0 {
				vertexCount = len(buf.Data) / buf.Size
			}
		}
	} else if shader.GetVertexCount() > 0 {
		// Legacy interleaved vertex data
		vertices := shader.GetVertices()
		vertexCount = shader.GetVertexCount()

		// Create VBO if needed
		vbo := shader.GetVBO()
		if vbo == 0 {
			vbo = uint32(p.ctx.CreateBuffer())
			shader.SetVBO(vbo)
		}
		p.ctx.BindBuffer(arrayBuffer, Buffer(vbo))
		if len(vertices) > 0 {
			p.ctx.BufferData(arrayBuffer, vertices, dynamicDraw)
		}

		// Calculate total stride and set up attributes
		totalStride := 0
		for _, attr := range attrs {
			totalStride += attr.size
		}

		attrOffset := 0
		for _, attr := range attrs {
			// Try stored location first, fall back to GL query
			storedLoc := shader.GetAttributeLocation(attr.name)
			var attribLoc Attribute
			if storedLoc >= 0 {
				attribLoc = Attribute(storedLoc)
			} else {
				attribLoc = p.ctx.GetAttribLocation(prog, attr.name)
			}
			if attribLoc >= 0 {
				p.ctx.EnableVertexAttribArray(attribLoc)
				enabledAttribs = append(enabledAttribs, attribLoc)
				strideBytes := 0
				if len(attrs) > 1 {
					strideBytes = totalStride * 4
				}
				p.ctx.VertexAttribPointerWithOffset(attribLoc, attr.size, float, false, strideBytes, attrOffset*4)
			}
			attrOffset += attr.size
		}
	} else {
		// Default fullscreen quad
		vertices := []float32{
			-1, -1,  // 0: bottom-left
			 1, -1,  // 1: bottom-right
			-1,  1,  // 2: top-left
			 1,  1,  // 3: top-right
		}
		vertexCount = 4

		vbo := shader.GetVBO()
		if vbo == 0 {
			vbo = uint32(p.ctx.CreateBuffer())
			shader.SetVBO(vbo)
		}
		p.ctx.BindBuffer(arrayBuffer, Buffer(vbo))
		p.ctx.BufferData(arrayBuffer, vertices, dynamicDraw)

		// Set up 'vert' attribute for fullscreen quad
		storedVertLoc := shader.GetAttributeLocation("vert")
		var vertAttrib Attribute
		if storedVertLoc >= 0 {
			vertAttrib = Attribute(storedVertLoc)
		} else {
			vertAttrib = p.ctx.GetAttribLocation(prog, "vert")
		}
		if vertAttrib >= 0 {
			p.ctx.EnableVertexAttribArray(vertAttrib)
			enabledAttribs = append(enabledAttribs, vertAttrib)
			p.ctx.VertexAttribPointerWithOffset(vertAttrib, 2, float, false, 0, 0)
		}
	}

	// Create and upload IBO if we have indices for indexed drawing
	useIndexedDrawing := false
	if indexCount > 0 && useSeparateBuffers {
		indices := shader.GetIndices()
		if len(indices) > 0 {
			// Create IBO if needed
			ibo := shader.GetIBO()
			if ibo == 0 {
				ibo = uint32(p.ctx.CreateBuffer())
				shader.SetIBO(ibo)
			}
			p.ctx.BindBuffer(elementArrayBuffer, Buffer(ibo))
			p.ctx.BufferDataUint16(elementArrayBuffer, indices, dynamicDraw)
			useIndexedDrawing = true
		}
	}

	p.logError()

	if len(enabledAttribs) == 0 && shaderDebug {
		log.Printf("[drawShader] WARNING: No vertex attributes bound!")
	}

	// Process render command queue (immediate-mode: set uniform → draw → set uniform → draw)
	renderCommands := shader.GetRenderCommands()
	if len(renderCommands) > 0 {
		// Set up GL state for 3D rendering (three.js style)
		p.ctx.Enable(depthTest)
		p.ctx.DepthFunc(lequal)
		p.ctx.DepthMask(true)
		p.ctx.Enable(cullFace)
		p.ctx.CullFace(backFace)
		p.ctx.FrontFace(ccw)
		p.ctx.Enable(programPointSize)

		if shaderDebug {
			drawCount := 0
			useProgramCount := 0
			for _, cmd := range renderCommands {
				if cmd.Type == "drawElements" || cmd.Type == "drawArrays" {
					drawCount++
				}
				if cmd.Type == "useProgram" {
					useProgramCount++
				}
			}
			log.Printf("[drawShader] Frame: %d commands, %d draws, %d usePrograms", len(renderCommands), drawCount, useProgramCount)
		}

		// Execute commands in order (three.js immediate-mode style)
		for cmdIdx := 0; cmdIdx < len(renderCommands); cmdIdx++ {
			cmd := renderCommands[cmdIdx]
			_ = cmdIdx
			switch cmd.Type {
			case "uniform":
				setUniform(cmd.Name, cmd.Value)
			case "drawElements":
				if params, ok := cmd.Value.(canvas.DrawElementsParams); ok {
					if params.Count > 0 {
						if shaderDebug {
							log.Printf("[drawShader] drawElements: count=%d mode=%d inst=%d attrs=%d indices=%d",
								params.Count, params.Mode, params.InstanceCount, len(params.AttributeBuffers), len(params.Indices))
							for name, buf := range params.AttributeBuffers {
								if buf != nil {
									log.Printf("[drawShader]   cmdAttr %s: %d floats, size=%d stride=%d offset=%d",
										name, len(buf.Data), buf.Size, buf.Stride, buf.Offset)
								}
							}
						}
						// Use captured geometry from the draw command if available
						cmdAttrBuffers := params.AttributeBuffers
						cmdIndices := params.Indices
						if len(cmdAttrBuffers) > 0 {
							// Re-upload geometry for this specific draw call
							for _, attr := range attrs {
								buf := cmdAttrBuffers[attr.name]
								if buf == nil || len(buf.Data) == 0 {
									continue
								}
								attribLoc := p.ctx.GetAttribLocation(prog, attr.name)
								if attribLoc < 0 {
									storedLoc := shader.GetAttributeLocation(attr.name)
									if storedLoc >= 0 && !mat4Locs[int(storedLoc)] {
										attribLoc = Attribute(storedLoc)
									}
								}
								if attribLoc < 0 {
									continue
								}
								// Create VBO if needed
								if buf.VBO == 0 {
									buf.VBO = uint32(p.ctx.CreateBuffer())
								}
								p.ctx.BindBuffer(arrayBuffer, Buffer(buf.VBO))
								p.ctx.BufferData(arrayBuffer, buf.Data, dynamicDraw)
								// Handle mat4: 4 consecutive vec4 columns sharing one VBO
								if attr.glslType == "mat4" {
									div := attribDivisors[attr.name]
									for col := 0; col < 4; col++ {
										loc := Attribute(int(attribLoc) + col)
										p.ctx.EnableVertexAttribArray(loc)
										p.ctx.VertexAttribPointerWithOffset(loc, 4, float, false, buf.Stride, buf.Offset+col*16)
										if div > 0 {
											p.ctx.VertexAttribDivisor(uint32(loc), div)
										}
									}
								} else {
									p.ctx.EnableVertexAttribArray(attribLoc)
									p.ctx.VertexAttribPointerWithOffset(attribLoc, buf.Size, float, false, buf.Stride, buf.Offset)
									// Set divisor for instanced attributes
									if div := attribDivisors[attr.name]; div > 0 {
										p.ctx.VertexAttribDivisor(uint32(attribLoc), div)
									}
								}
							}
						}
						if len(cmdIndices) > 0 {
							// Re-upload index buffer for this draw call
							ibo := shader.GetIBO()
							if ibo == 0 {
								ibo = uint32(p.ctx.CreateBuffer())
								shader.SetIBO(ibo)
							}
							p.ctx.BindBuffer(elementArrayBuffer, Buffer(ibo))
							p.ctx.BufferDataUint16(elementArrayBuffer, cmdIndices, dynamicDraw)
							if params.InstanceCount > 0 {
								p.ctx.DrawElementsInstanced(params.Mode, int32(params.Count), unsignedShort, params.Offset, int32(params.InstanceCount))
							} else {
								p.ctx.DrawElements(params.Mode, int32(params.Count), unsignedShort, params.Offset)
							}
						} else if useIndexedDrawing {
							// Fall back to pre-uploaded indices
							if params.InstanceCount > 0 {
								p.ctx.DrawElementsInstanced(params.Mode, int32(params.Count), unsignedShort, params.Offset, int32(params.InstanceCount))
							} else {
								p.ctx.DrawElements(params.Mode, int32(params.Count), unsignedShort, params.Offset)
							}
						}
					}
				}
			case "drawArrays":
				if params, ok := cmd.Value.(canvas.DrawArraysParams); ok {
					if params.Count > 0 {
						// Use captured geometry from the draw command if available
						cmdAttrBuffers := params.AttributeBuffers
						if len(cmdAttrBuffers) > 0 {
							// Re-upload geometry for this specific draw call
							for _, attr := range attrs {
								buf := cmdAttrBuffers[attr.name]
								if buf == nil || len(buf.Data) == 0 {
									continue
								}
								attribLoc := p.ctx.GetAttribLocation(prog, attr.name)
								if attribLoc < 0 {
									storedLoc := shader.GetAttributeLocation(attr.name)
									if storedLoc >= 0 && !mat4Locs[int(storedLoc)] {
										attribLoc = Attribute(storedLoc)
									}
								}
								if attribLoc < 0 {
									continue
								}
								// Create VBO if needed
								if buf.VBO == 0 {
									buf.VBO = uint32(p.ctx.CreateBuffer())
								}
								p.ctx.BindBuffer(arrayBuffer, Buffer(buf.VBO))
								p.ctx.BufferData(arrayBuffer, buf.Data, dynamicDraw)
								// Handle mat4: 4 consecutive vec4 columns sharing one VBO
								if attr.glslType == "mat4" {
									div := attribDivisors[attr.name]
									for col := 0; col < 4; col++ {
										loc := Attribute(int(attribLoc) + col)
										p.ctx.EnableVertexAttribArray(loc)
										p.ctx.VertexAttribPointerWithOffset(loc, 4, float, false, buf.Stride, buf.Offset+col*16)
										if div > 0 {
											p.ctx.VertexAttribDivisor(uint32(loc), div)
										}
									}
								} else {
									p.ctx.EnableVertexAttribArray(attribLoc)
									p.ctx.VertexAttribPointerWithOffset(attribLoc, buf.Size, float, false, buf.Stride, buf.Offset)
									// Set divisor for instanced attributes
									if div := attribDivisors[attr.name]; div > 0 {
										p.ctx.VertexAttribDivisor(uint32(attribLoc), div)
									}
								}
							}
						}
						if params.InstanceCount > 0 {
							p.ctx.DrawArraysInstanced(params.Mode, int32(params.First), int32(params.Count), int32(params.InstanceCount))
						} else {
							p.ctx.DrawArrays(params.Mode, params.First, params.Count)
						}
					}
				}
			case "clearColor":
				if cc, ok := cmd.Value.([4]float32); ok {
					p.ctx.ClearColor(cc[0], cc[1], cc[2], cc[3])
				}
			case "clear":
				if mask, ok := cmd.Value.(uint32); ok {
					// Re-apply stored clear color before clearing
					// Fyne's painter.Clear() resets glClearColor to the theme bg each frame,
					// but Three.js only sends clearColor when the value changes (WebGLState caching)
					cc := shader.GetClearColor()
					p.ctx.ClearColor(cc[0], cc[1], cc[2], cc[3])
					p.ctx.Clear(mask)
				} else {
					// Legacy path (QueueClear without mask)
					cc := shader.GetClearColor()
					p.ctx.ClearColor(cc[0], cc[1], cc[2], cc[3])
					p.ctx.Clear(colorBufferBit | depthBufferBit)
				}
			case "enable":
				if cap, ok := cmd.Value.(uint32); ok {
					p.ctx.Enable(cap)
				}
			case "disable":
				if cap, ok := cmd.Value.(uint32); ok {
					p.ctx.Disable(cap)
				}
			case "cullFace":
				if mode, ok := cmd.Value.(uint32); ok {
					p.ctx.CullFace(mode)
				}
			case "blendFunc":
				if factors, ok := cmd.Value.([2]uint32); ok {
					p.ctx.BlendFunc(factors[0], factors[1])
				}
			case "useProgram":
				if params, ok := cmd.Value.(canvas.UseProgramParams); ok {
					cacheKey := params.VertexSource + "|||" + params.FragmentSource
					cachedProg, found := shader.GetProgramCache()[cacheKey]
					if found && cachedProg != 0 {
						prog = Program(cachedProg)
					} else {
						newProg := p.compileShaderProgram(params.VertexSource, params.FragmentSource)
						if newProg != 0 {
							shader.SetProgramCache(cacheKey, uint32(newProg))
							prog = newProg
						}
					}
					p.ctx.UseProgram(prog)
					attrs = parseVertexAttributes(params.VertexSource)
					for k := range shader.UniformLocs() {
						delete(shader.UniformLocs(), k)
					}
				}

			// ═══ FBO render commands ═══
			case "createFramebuffer":
				jsId, _ := strconv.Atoi(cmd.Name)
				fb := p.ctx.CreateFramebuffer()
				shader.SetFBOCache(jsId, uint32(fb))
				if shaderDebug {
					log.Printf("[drawShader] createFramebuffer: js=%d -> gl=%d", jsId, fb)
				}

			case "deleteFramebuffer":
				jsId, _ := strconv.Atoi(cmd.Name)
				glId := shader.GetFBOCache(jsId)
				if glId != 0 {
					p.ctx.DeleteFramebuffer(Framebuffer(glId))
					shader.DeleteFBOCache(jsId)
				}

			case "bindFramebuffer":
				if params, ok := cmd.Value.(canvas.BindFramebufferParams); ok {
					if params.FramebufferId == 0 {
						p.ctx.BindFramebuffer(params.Target, Framebuffer(0))
					} else {
						glId := shader.GetFBOCache(params.FramebufferId)
						p.ctx.BindFramebuffer(params.Target, Framebuffer(glId))
					}
				}

			case "framebufferTexture2D":
				if params, ok := cmd.Value.(canvas.FramebufferTexture2DParams); ok {
					// Look up GL texture ID - check GPU tex cache first, then image-based textures
					glTexId := shader.GetGPUTexCache(params.TextureId)
					if glTexId == 0 {
						// Try the image-based texture cache
						for texName := range shader.GetTextures() {
							if cached, ok := shader.GetTextureCache()[texName]; ok {
								// This is a heuristic - we match by checking if the texture ID maps
								// In practice, shadow map textures go through gpuTexCache
								_ = cached
							}
						}
					}
					if glTexId != 0 {
						p.ctx.FramebufferTexture2D(params.Target, params.Attachment, params.Textarget, Texture(glTexId), params.Level)
					} else if shaderDebug {
						log.Printf("[drawShader] framebufferTexture2D: texture %d not found in cache", params.TextureId)
					}
				}

			case "createRenderbuffer":
				jsId, _ := strconv.Atoi(cmd.Name)
				rb := p.ctx.CreateRenderbuffer()
				shader.SetRBOCache(jsId, uint32(rb))
				if shaderDebug {
					log.Printf("[drawShader] createRenderbuffer: js=%d -> gl=%d", jsId, rb)
				}

			case "deleteRenderbuffer":
				jsId, _ := strconv.Atoi(cmd.Name)
				glId := shader.GetRBOCache(jsId)
				if glId != 0 {
					p.ctx.DeleteRenderbuffer(Renderbuffer(glId))
					shader.DeleteRBOCache(jsId)
				}

			case "bindRenderbuffer":
				if params, ok := cmd.Value.(canvas.BindFramebufferParams); ok {
					if params.FramebufferId == 0 {
						p.ctx.BindRenderbuffer(params.Target, Renderbuffer(0))
					} else {
						glId := shader.GetRBOCache(params.FramebufferId)
						p.ctx.BindRenderbuffer(params.Target, Renderbuffer(glId))
					}
				}

			case "renderbufferStorage":
				if params, ok := cmd.Value.(canvas.RenderbufferStorageParams); ok {
					p.ctx.RenderbufferStorage(params.Target, params.Internalformat, params.Width, params.Height)
				}

			case "framebufferRenderbuffer":
				if params, ok := cmd.Value.(canvas.FramebufferRenderbufferParams); ok {
					glId := shader.GetRBOCache(params.RenderbufferId)
					p.ctx.FramebufferRenderbuffer(params.Target, params.Attachment, params.RenderbufferTarget, Renderbuffer(glId))
				}

			case "viewport":
				if params, ok := cmd.Value.(canvas.ViewportParams); ok {
					p.ctx.Viewport(params.X, params.Y, params.Width, params.Height)
				}

			case "colorMask":
				if params, ok := cmd.Value.(canvas.ColorMaskParams); ok {
					p.ctx.ColorMask(params.R, params.G, params.B, params.A)
				}

			case "clearDepth":
				if params, ok := cmd.Value.(canvas.ClearDepthParams); ok {
					p.ctx.ClearDepthf(params.Depth)
				}

			case "drawBuffers":
				if params, ok := cmd.Value.(canvas.DrawBuffersParams); ok {
					p.ctx.DrawBuffers(params.Buffers)
				}

			case "texImage2D_gpu":
				// GPU-only texture allocation (e.g., shadow maps with null pixels, DataTexture storage)
				// IMPORTANT: Use scratch texture unit 15 to avoid corrupting sampler bindings
				// set up in Phase 2.1. Without this, BindTexture here overwrites whatever
				// texture unit is active, making Phase 2.1 textures invisible.
				jsTexId, _ := strconv.Atoi(cmd.Name)
				if params, ok := cmd.Value.(canvas.TexImage2DGPUParams); ok {
					// Skip if Phase 2.1 already created and uploaded a texture for this JS ID.
					// This prevents overwriting a real texture with an empty one when
					// texStorage2D render commands execute after Phase 2.1.
					if existingGlId := shader.GetGPUTexCache(jsTexId); existingGlId != 0 {
						if shaderDebug {
							log.Printf("[drawShader] texImage2D_gpu: SKIPPED js=%d (already have gl=%d)", jsTexId, existingGlId)
						}
					} else {
						p.ctx.ActiveTexture(texture0 + 15) // scratch unit
						tex := p.ctx.CreateTexture()
						p.ctx.BindTexture(params.Target, tex)
						// Set appropriate filtering and wrapping
						p.ctx.TexParameteri(params.Target, textureMinFilter, nearest)
						p.ctx.TexParameteri(params.Target, textureMagFilter, nearest)
						p.ctx.TexParameteri(params.Target, textureWrapS, clampToEdge)
						p.ctx.TexParameteri(params.Target, textureWrapT, clampToEdge)
						// Allocate texture storage without data
						p.ctx.TexImage2DEmpty(params.Target, params.Level, params.Internalformat, params.Width, params.Height, params.Format, params.Type)
						p.ctx.BindTexture(params.Target, Texture(0)) // unbind from scratch unit
						p.ctx.ActiveTexture(texture0) // restore to unit 0
						shader.SetGPUTexCache(jsTexId, uint32(tex))
						if shaderDebug {
							log.Printf("[drawShader] texImage2D_gpu: js=%d -> gl=%d (%dx%d, fmt=0x%x)", jsTexId, tex, params.Width, params.Height, params.Format)
						}
					}
				}

			case "texParameteri":
				if params, ok := cmd.Value.(canvas.TexParameteriParams); ok {
					p.ctx.TexParameteri(params.Target, params.Pname, params.Param)
				}

			case "activeTexture":
				if texEnum, ok := cmd.Value.(uint32); ok {
					p.ctx.ActiveTexture(texEnum)
				}

			case "bindTexture":
				jsTexId, _ := strconv.Atoi(cmd.Name)
				target, _ := cmd.Value.(uint32)
				if jsTexId == 0 {
					// Unbind
					p.ctx.BindTexture(target, Texture(0))
				} else {
					// Check GPU texture cache first (shadow maps etc.)
					if glId := shader.GetGPUTexCache(jsTexId); glId != 0 {
						p.ctx.BindTexture(target, Texture(glId))
						// On-demand upload: if GPU texture was allocated empty (texStorage2D)
						// and we have CPU image data, upload it now
						if !shader.IsGPUTexUploaded(jsTexId) {
							cpuImg := shader.GetCPUTexImage(jsTexId)
							if cpuImg != nil {
								rgba := p.textureToRGBA(cpuImg)
								if rgba != nil && len(rgba.Pix) > 0 {
									p.ctx.TexParameteri(target, textureMinFilter, linear)
									p.ctx.TexParameteri(target, textureMagFilter, linear)
									texFmt := shader.GetGPUTexFormat(jsTexId)
									w := rgba.Rect.Size().X
									h := rgba.Rect.Size().Y
									if texFmt.Format == 0x1903 { // GL_RED
										// Extract R channel only (1 byte/pixel)
										redData := make([]byte, w*h)
										for i := 0; i < w*h; i++ {
											redData[i] = rgba.Pix[i*4]
										}
										p.ctx.TexImage2D(target, 0, w, h, texFmt.Format, unsignedByte, redData)
									} else {
										p.ctx.TexImage2D(target, 0, w, h, colorFormatRGBA, unsignedByte, rgba.Pix)
									}
									shader.SetGPUTexUploaded(jsTexId)
									if shaderDebug {
										log.Printf("[drawShader] bindTexture: ON-DEMAND uploaded %dx%d to js=%d gl=%d (fmt=0x%x)", w, h, jsTexId, glId, texFmt.Format)
									}
								}
							}
						}
					} else if glId := shader.GetTextureCache()[fmt.Sprintf("tex_%d", jsTexId)]; glId != 0 {
						// Fall back to regular texture cache
						p.ctx.BindTexture(target, Texture(glId))
					}
				}

			case "depthFunc":
				if fn, ok := cmd.Value.(uint32); ok {
					p.ctx.DepthFunc(fn)
				}

			case "depthMask":
				if flag, ok := cmd.Value.(bool); ok {
					p.ctx.DepthMask(flag)
				}

			case "frontFace":
				if mode, ok := cmd.Value.(uint32); ok {
					p.ctx.FrontFace(mode)
				}

			case "blendFuncSeparate":
				if factors, ok := cmd.Value.([4]uint32); ok {
					p.ctx.BlendFuncSeparate(factors[0], factors[1], factors[2], factors[3])
				}
			case "blendEquation":
				if mode, ok := cmd.Value.(uint32); ok {
					p.ctx.BlendEquation(mode)
				}
			case "blendEquationSeparate":
				if modes, ok := cmd.Value.([2]uint32); ok {
					p.ctx.BlendEquationSeparate(modes[0], modes[1])
				}
			case "blendColor":
				if c, ok := cmd.Value.([4]float32); ok {
					p.ctx.BlendColor(c[0], c[1], c[2], c[3])
				}
			case "polygonOffset":
				if params, ok := cmd.Value.([2]float32); ok {
					p.ctx.PolygonOffset(params[0], params[1])
				}
			case "lineWidth":
				if w, ok := cmd.Value.(float32); ok {
					p.ctx.LineWidth(w)
				}
			case "scissor":
				if rect, ok := cmd.Value.([4]int32); ok {
					p.ctx.Scissor(rect[0], rect[1], rect[2], rect[3])
				}

			case "vertexAttrib1f", "vertexAttrib1fv":
				if packed, ok := cmd.Value.([]float32); ok && len(packed) >= 2 {
					p.ctx.VertexAttrib1f(uint32(packed[0]), packed[1])
				}
			case "vertexAttrib2f", "vertexAttrib2fv":
				if packed, ok := cmd.Value.([]float32); ok && len(packed) >= 3 {
					p.ctx.VertexAttrib2f(uint32(packed[0]), packed[1], packed[2])
				}
			case "vertexAttrib3f", "vertexAttrib3fv":
				if packed, ok := cmd.Value.([]float32); ok && len(packed) >= 4 {
					p.ctx.VertexAttrib3f(uint32(packed[0]), packed[1], packed[2], packed[3])
				}
			case "vertexAttrib4f", "vertexAttrib4fv":
				if packed, ok := cmd.Value.([]float32); ok && len(packed) >= 5 {
					p.ctx.VertexAttrib4f(uint32(packed[0]), packed[1], packed[2], packed[3], packed[4])
				}
			case "vertexAttribDivisor":
				if packed, ok := cmd.Value.([2]uint32); ok {
					p.ctx.VertexAttribDivisor(packed[0], packed[1])
				}
			case "stencilFunc":
				if params, ok := cmd.Value.([3]uint32); ok {
					p.ctx.StencilFunc(params[0], int32(params[1]), params[2])
				}
			case "stencilOp":
				if params, ok := cmd.Value.([3]uint32); ok {
					p.ctx.StencilOp(params[0], params[1], params[2])
				}
			case "stencilMask":
				if mask, ok := cmd.Value.(uint32); ok {
					p.ctx.StencilMask(mask)
				}
			case "stencilFuncSeparate":
				if params, ok := cmd.Value.([4]uint32); ok {
					p.ctx.StencilFuncSeparate(params[0], params[1], int32(params[2]), params[3])
				}
			case "stencilOpSeparate":
				if params, ok := cmd.Value.([4]uint32); ok {
					p.ctx.StencilOpSeparate(params[0], params[1], params[2], params[3])
				}
			case "stencilMaskSeparate":
				if params, ok := cmd.Value.([2]uint32); ok {
					p.ctx.StencilMaskSeparate(params[0], params[1])
				}
			case "clearStencil":
				if val, ok := cmd.Value.(int32); ok {
					p.ctx.ClearStencil(val)
				}
			}
		}

		// Restore GL state
		p.ctx.BindFramebuffer(framebuffer, Framebuffer(0)) // Unbind any FBO
		p.ctx.ColorMask(true, true, true, true) // Restore color mask
		p.ctx.Disable(depthTest)
		p.ctx.Disable(cullFace)
		p.ctx.Disable(programPointSize)
	} else {
		// Fallback: single draw (for simple shaders without command queue)
		if useIndexedDrawing && indexCount > 0 {
			p.ctx.DrawElements(triangles, int32(indexCount), unsignedShort, 0)
		} else if vertexCount > 0 {
			if useSeparateBuffers || shader.GetVertexCount() > 0 {
				p.ctx.DrawArrays(triangles, 0, vertexCount)
			} else {
				p.ctx.DrawArrays(triangleStrip, 0, vertexCount)
			}
		}
	}
	p.logError()

	// Disable enabled vertex attributes
	for _, attrib := range enabledAttribs {
		p.ctx.DisableVertexAttribArray(attrib)
	}

	// Unbind and delete the shader VAO, restore VAO 0 for Fyne's default rendering
	p.ctx.BindVertexArray(0)
	p.ctx.DeleteVertexArray(shaderVAO)

	// Restore original viewport (full frame)
	frameW := int(p.pixScale * frame.Width)
	frameH := int(p.pixScale * frame.Height)
	p.ctx.Viewport(0, 0, frameW, frameH)

	// Restore default program
	p.ctx.UseProgram(p.program.ref)

	// Re-enable default program vertex attribute arrays
	// (shader's DisableVertexAttribArray may have disabled them if indices match)
	p.ctx.EnableVertexAttribArray(p.program.attributes["vert"])
	p.ctx.EnableVertexAttribArray(p.program.attributes["vertTexCoord"])

	// Signal that paint is complete (for sync with JS frame loop)
	shader.SignalPaintDone()
}
