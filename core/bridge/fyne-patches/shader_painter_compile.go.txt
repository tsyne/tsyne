package gl

import (
	"log"
	"os"
	"regexp"
	"strings"
)

// attrRegex matches attribute declarations in GLSL vertex shaders, including:
// - Standard:  attribute vec3 position; / in vec3 position;
// - Layout:    layout(location=0)in vec3 position;  (minified, no space before 'in')
// - Layout:    layout(location = 0) in vec3 position;  (formatted)
// Uses non-anchored matching so it works on single-line minified shaders.
var attrRegex = regexp.MustCompile(`(?:layout\s*\(\s*location\s*=\s*(\d+)\s*\)\s*)?(?:attribute|in)\s+(vec2|vec3|vec4|float|mat2|mat3|mat4)\s+(\w+)\s*;`)

// parsedAttrsCache caches parsed vertex attributes keyed by vertex source.
// Capped at 64 entries to prevent unbounded growth.
var parsedAttrsCache = make(map[string][]shaderAttribute)
const maxParsedAttrsCache = 64

// parseVertexAttributes extracts attribute declarations from a GLSL vertex shader
// Returns a list of attributes with their names and sizes.
// Handles both formatted and minified shaders, including layout(location=N) qualifiers.
// Results are cached by vertex source string.
func parseVertexAttributes(vertexSrc string) []shaderAttribute {
	if cached, ok := parsedAttrsCache[vertexSrc]; ok {
		return cached
	}

	var attrs []shaderAttribute

	matches := attrRegex.FindAllStringSubmatch(vertexSrc, -1)
	for _, match := range matches {
		if len(match) >= 4 {
			// match[1] = layout location (may be empty)
			// match[2] = GLSL type
			// match[3] = attribute name
			glslType := match[2]
			name := match[3]
			size := getTypeSize(glslType)
			attr := shaderAttribute{
				name:     name,
				glslType: glslType,
				size:     size,
			}
			// Store layout location if present (-1 = no layout)
			attr.layoutLocation = -1
			if match[1] != "" {
				loc := 0
				for _, ch := range match[1] {
					loc = loc*10 + int(ch-'0')
				}
				attr.layoutLocation = loc
			}
			attrs = append(attrs, attr)
		}
	}

	// Evict entire cache if over limit (simple reset, not LRU — these are cheap to reparse)
	if len(parsedAttrsCache) >= maxParsedAttrsCache {
		parsedAttrsCache = make(map[string][]shaderAttribute)
	}
	parsedAttrsCache[vertexSrc] = attrs
	return attrs
}

// getTypeSize returns the number of float components for a GLSL type
func getTypeSize(glslType string) int {
	switch glslType {
	case "float":
		return 1
	case "vec2":
		return 2
	case "vec3":
		return 3
	case "vec4":
		return 4
	case "mat2":
		return 4  // 2x2 = 4 floats
	case "mat3":
		return 9  // 3x3 = 9 floats
	case "mat4":
		return 16 // 4x4 = 16 floats
	default:
		return 0
	}
}

// compileShaderProgram compiles and links a vertex and fragment shader
func (p *painter) compileShaderProgram(vertexSrc, fragmentSrc string) Program {
	// Prepend standard uniforms to fragment shader if it doesn't have a version
	if !strings.HasPrefix(strings.TrimSpace(fragmentSrc), "#version") {
		fragmentSrc = getShaderFragmentPrefix() + fragmentSrc
	}

	// Write shaders to disk for debugging on GLES
	if shaderDebug && len(vertexSrc) > 500 {
		os.WriteFile("/tmp/tsyne_compile_vert.glsl", []byte(vertexSrc), 0644)
		os.WriteFile("/tmp/tsyne_compile_frag.glsl", []byte(fragmentSrc), 0644)
		v := vertexSrc; if len(v) > 200 { v = v[:200] }
		f := fragmentSrc; if len(f) > 200 { f = f[:200] }
		log.Printf("[shader] compiling vert(%d bytes): %s...", len(vertexSrc), strings.ReplaceAll(v, "\n", "\\n"))
		log.Printf("[shader] compiling frag(%d bytes): %s...", len(fragmentSrc), strings.ReplaceAll(f, "\n", "\\n"))
	}

	// Compile vertex shader
	vShader := p.ctx.CreateShader(vertexShader)
	p.ctx.ShaderSource(vShader, vertexSrc)
	p.ctx.CompileShader(vShader)
	if p.ctx.GetShaderi(vShader, compileStatus) == 0 {
		logMsg := p.ctx.GetShaderInfoLog(vShader)
		log.Printf("[shader] Vertex shader compile error: %s", logMsg)
		return 0
	}

	// Compile fragment shader
	fShader := p.ctx.CreateShader(fragmentShader)
	p.ctx.ShaderSource(fShader, fragmentSrc)
	p.ctx.CompileShader(fShader)
	if p.ctx.GetShaderi(fShader, compileStatus) == 0 {
		logMsg := p.ctx.GetShaderInfoLog(fShader)
		log.Printf("[shader] Fragment shader compile error: %s", logMsg)
		return 0
	}

	// Link program
	program := p.ctx.CreateProgram()
	p.ctx.AttachShader(program, vShader)
	p.ctx.AttachShader(program, fShader)
	p.ctx.LinkProgram(program)
	if p.ctx.GetProgrami(program, linkStatus) == 0 {
		logMsg := p.ctx.GetProgramInfoLog(program)
		log.Printf("[shader] Program link error: %s", logMsg)
		return 0
	}

	// Note: Individual shaders could be deleted here but Fyne's context doesn't expose DeleteShader.
	// The GL resources are cleaned up when the program is deleted.

	return program
}
