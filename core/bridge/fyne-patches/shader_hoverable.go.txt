// attrMapPool pools maps used for attribute buffer snapshots in draw calls.
var attrMapPool = sync.Pool{
	New: func() interface{} {
		m := make(map[string]*AttributeBuffer, 8)
		return &m
	},
}

// attrBufPool pools AttributeBuffer structs used in draw call snapshots.
var attrBufPool = sync.Pool{
	New: func() interface{} {
		return &AttributeBuffer{}
	},
}

// snapshotAttributeBuffers creates a pooled snapshot of the current attribute buffers.
// The returned map and its AttributeBuffer values come from sync.Pool and must be
// returned via RecycleAttrSnapshot after the painter consumes them.
func (s *Shader) snapshotAttributeBuffers() map[string]*AttributeBuffer {
	mp := attrMapPool.Get().(*map[string]*AttributeBuffer)
	m := *mp
	// Clear any leftover entries from previous use (Go 1.19 compat — no clear())
	for k := range m {
		delete(m, k)
	}
	for name, buf := range s.AttributeBuffers {
		if buf != nil && len(buf.Data) > 0 {
			ab := attrBufPool.Get().(*AttributeBuffer)
			ab.Data = buf.Data
			ab.Size = buf.Size
			ab.VBO = buf.VBO
			ab.Stride = buf.Stride
			ab.Offset = buf.Offset
			ab.Generation = buf.Generation
			m[name] = ab
		}
	}
	return m
}

// RecycleAttrSnapshot returns a snapshot map and its AttributeBuffer entries to
// the pool. Called by the painter after consuming a draw command.
func RecycleAttrSnapshot(m map[string]*AttributeBuffer) {
	if m == nil {
		return
	}
	for k, buf := range m {
		if buf != nil {
			// Zero out slice reference to avoid pinning data
			buf.Data = nil
			attrBufPool.Put(buf)
		}
		delete(m, k)
	}
	attrMapPool.Put(&m)
}

// QueueRenderCommand adds a render command to the queue.
// Commands are executed in order during the paint cycle.
func (s *Shader) QueueRenderCommand(cmd RenderCommand) {
	s.renderMu.Lock()
	s.renderCommands = append(s.renderCommands, cmd)
	s.renderMu.Unlock()
}

// QueueUniform queues a uniform set command.
func (s *Shader) QueueUniform(name string, value interface{}) {
	s.QueueRenderCommand(RenderCommand{
		Type:  "uniform",
		Name:  name,
		Value: value,
	})
}

// QueueDrawElements queues a drawElements call with a snapshot of current geometry.
// Safe to share slice references: glBufferData always allocates new slices,
// and glBufferSubData uses copy-on-write.
func (s *Shader) QueueDrawElements(mode uint32, count int, offset int) {
	s.attrBuffersMu.RLock()
	attrSnapshot := s.snapshotAttributeBuffers()
	s.attrBuffersMu.RUnlock()

	tag := s.CurrentDrawTag
	s.CurrentDrawTag = ""
	s.QueueRenderCommand(RenderCommand{
		Type: "drawElements",
		Name: tag,
		Value: DrawElementsParams{
			Mode:             mode,
			Count:            count,
			Offset:           offset,
			AttributeBuffers: attrSnapshot,
			Indices:          s.Indices,
		},
	})
}

// QueueDrawArrays queues a drawArrays call with a snapshot of current geometry.
// Safe to share slice references: glBufferData always allocates new slices,
// and glBufferSubData uses copy-on-write.
func (s *Shader) QueueDrawArrays(mode uint32, first int, count int) {
	s.attrBuffersMu.RLock()
	attrSnapshot := s.snapshotAttributeBuffers()
	s.attrBuffersMu.RUnlock()

	tag := s.CurrentDrawTag
	s.CurrentDrawTag = ""
	s.QueueRenderCommand(RenderCommand{
		Type: "drawArrays",
		Name: tag,
		Value: DrawArraysParams{
			Mode:             mode,
			First:            first,
			Count:            count,
			AttributeBuffers: attrSnapshot,
		},
	})
}

// QueueClear queues a clear command.
func (s *Shader) QueueClear() {
	s.QueueRenderCommand(RenderCommand{
		Type: "clear",
	})
}

// GetRenderCommands returns the render command queue.
// If new commands have been queued since the last call, they replace the
// previous frame's commands. Otherwise the previous frame is replayed so
// that Fyne repaints don't produce a black flash between animation ticks.
// Sets LastGetWasNew so callers can distinguish new frames from replays.
func (s *Shader) GetRenderCommands() []RenderCommand {
	s.renderMu.Lock()
	defer s.renderMu.Unlock()
	if len(s.renderCommands) > 0 {
		// New frame — recycle pooled attribute snapshots from the old frame's
		// commands before we lose the reference. These were already consumed
		// by the painter in the previous paint cycle.
		old := s.lastRenderCommands
		for _, cmd := range old {
			switch cmd.Type {
			case "drawElements":
				if params, ok := cmd.Value.(DrawElementsParams); ok {
					RecycleAttrSnapshot(params.AttributeBuffers)
				}
			case "drawArrays":
				if params, ok := cmd.Value.(DrawArraysParams); ok {
					RecycleAttrSnapshot(params.AttributeBuffers)
				}
			}
		}
		// Swap slices so the old lastRenderCommands backing array gets reused.
		// The painter fully consumes lastRenderCommands before the next
		// GetRenderCommands() call.
		s.lastRenderCommands = s.renderCommands
		if old != nil {
			s.renderCommands = old[:0] // reuse capacity
		} else {
			s.renderCommands = nil
		}
		s.LastGetWasNew = true
		return s.lastRenderCommands
	}
	// No new commands — replay the previous frame
	s.LastGetWasNew = false
	return s.lastRenderCommands
}

// HasRenderCommands returns true if there are queued render commands.
func (s *Shader) HasRenderCommands() bool {
	s.renderMu.Lock()
	defer s.renderMu.Unlock()
	return len(s.renderCommands) > 0
}

// HasRenderHistory returns true if there are render commands (new or replayed).
// Used to determine whether the render command loop handles texture/uniform setup,
// so Phase 2.1/2.2 (legacy texture setup) can be skipped.
func (s *Shader) HasRenderHistory() bool {
	s.renderMu.Lock()
	defer s.renderMu.Unlock()
	return len(s.renderCommands) > 0 || len(s.lastRenderCommands) > 0
}

// BeginPaint prepares for a paint cycle - must call WaitForPaint after Refresh.
func (s *Shader) BeginPaint() {
	s.paintMu.Lock()
	defer s.paintMu.Unlock()
	// Create a new channel for this paint cycle
	s.paintDone = make(chan struct{})
	s.paintPending = true
}

// SignalPaintDone is called by the painter after rendering completes.
func (s *Shader) SignalPaintDone() {
	s.paintMu.Lock()
	defer s.paintMu.Unlock()
	if s.paintPending && s.paintDone != nil {
		close(s.paintDone)
		s.paintPending = false
	}
}

// WaitForPaint blocks until the paint cycle completes (with timeout).
func (s *Shader) WaitForPaint() {
	s.paintMu.Lock()
	ch := s.paintDone
	pending := s.paintPending
	s.paintMu.Unlock()

	if !pending || ch == nil {
		return
	}

	// Wait for paint to complete, with a generous timeout.
	// CRITICAL: A short timeout (e.g., 16ms) causes the caller to return early
	// and send the next batch before the current paint finishes. The new batch's
	// render commands APPEND to the queue, doubling the GPU work per paint cycle.
	// This creates a positive feedback loop: more commands → longer paint →
	// more timeouts → more coalescing → GPU falls behind → machine freeze.
	// 5 seconds is long enough for even the worst paint cycle but short enough
	// to recover from a truly hung GPU.
	select {
	case <-ch:
		// Paint completed
	case <-time.After(5 * time.Second):
		log.Printf("[PAINT_WAIT_TIMEOUT] Paint did not complete within 5s — possible GPU stall")
	}
}

// RequestCapture sets up a capture-on-next-paint request.
// Returns a channel that will receive the captured pixel data.
// The painter calls ReadPixels at the end of drawShader, before SwapBuffers.
func (s *Shader) RequestCapture() <-chan CaptureResult {
	s.captureMu.Lock()
	defer s.captureMu.Unlock()
	ch := make(chan CaptureResult, 1)
	s.captureChan = ch
	return ch
}

// GetCaptureChan returns and clears the pending capture channel (called by painter).
func (s *Shader) GetCaptureChan() chan CaptureResult {
	s.captureMu.Lock()
	defer s.captureMu.Unlock()
	ch := s.captureChan
	s.captureChan = nil
	return ch
}

// RequestReadPixels stores a readPixels request and returns a channel for the result.
// The painter will execute glReadPixels on the GL thread and send the result.
func (s *Shader) RequestReadPixels(x, y, w, h int) <-chan ReadPixelsResult {
	s.readPixelsMu.Lock()
	defer s.readPixelsMu.Unlock()
	ch := make(chan ReadPixelsResult, 1)
	s.readPixelsChan = ch
	s.readPixelsReq = &ReadPixelsRequest{X: x, Y: y, Width: w, Height: h}
	return ch
}

// GetReadPixelsReq returns and clears the pending readPixels request (called by painter).
func (s *Shader) GetReadPixelsReq() (*ReadPixelsRequest, chan ReadPixelsResult) {
	s.readPixelsMu.Lock()
	defer s.readPixelsMu.Unlock()
	req := s.readPixelsReq
	ch := s.readPixelsChan
	s.readPixelsReq = nil
	s.readPixelsChan = nil
	return req, ch
}

// SetClearColor sets the clear color for glClear operations.
func (s *Shader) SetClearColor(r, g, b, a float32) {
	s.clearColor = [4]float32{r, g, b, a}
}

// GetClearColor returns the clear color.
func (s *Shader) GetClearColor() [4]float32 {
	return s.clearColor
}

// MouseEventCallback is a callback for mouse position events
type MouseEventCallback func(x, y float32)

// MouseButtonCallback is a callback for mouse button events (down/up)
type MouseButtonCallback func(x, y float32, button int)

// KeyCallback is a callback for keyboard events
type KeyCallback func(key string)

// ScrollCallback is a callback for scroll events
type ScrollCallback func(dx, dy float32)

// DragCallback is a callback for drag events
type DragCallback func(dx, dy float32)

// HoverableShader is a Shader that can receive mouse hover, click, keyboard,
// scroll, and drag events.
// Implements desktop.Hoverable, desktop.Mouseable, fyne.Focusable,
// desktop.Keyable, fyne.Scrollable, and fyne.Draggable.
type HoverableShader struct {
	*Shader
	onMouseIn    MouseEventCallback
	onMouseMoved MouseEventCallback
	onMouseOut   func()
	onMouseDown  MouseButtonCallback
	onMouseUp    MouseButtonCallback
	onKeyDown    KeyCallback
	onKeyUp      KeyCallback
	onScrolled   ScrollCallback
	onDragged    DragCallback
	onDragEnd    func()
	isHovered    bool
	focused      bool
	keysPressed  map[string]bool

	// Pointer lock state — when enabled, mouse deltas are delivered directly
	// from the GLFW cursor callback (bypassing Fyne's hit-testing) via
	// the onDragged callback. This emulates the browser's requestPointerLock() API.
	pointerLocked bool
}

// NewHoverableShader creates a new hoverable shader canvas object.
func NewHoverableShader(width, height float32, fragmentSrc string) *HoverableShader {
	return &HoverableShader{
		Shader:      NewShader(width, height, fragmentSrc),
		isHovered:   false,
		keysPressed: make(map[string]bool),
	}
}

// Refresh causes the shader to be redrawn.
// This overrides the embedded Shader's Refresh to ensure the HoverableShader wrapper is refreshed.
func (h *HoverableShader) Refresh() {
	h.Shader.cachedTime = float32(time.Since(h.Shader.startTime).Seconds())
	Refresh(h) // Refresh the HoverableShader, not the embedded Shader
}

// Verify HoverableShader implements all event interfaces
var _ desktop.Hoverable = (*HoverableShader)(nil)
var _ desktop.Mouseable = (*HoverableShader)(nil)
var _ fyne.Focusable = (*HoverableShader)(nil)
var _ desktop.Keyable = (*HoverableShader)(nil)
var _ fyne.Scrollable = (*HoverableShader)(nil)
var _ fyne.Draggable = (*HoverableShader)(nil)

// MouseIn is called when the mouse enters the widget
func (h *HoverableShader) MouseIn(event *desktop.MouseEvent) {
	h.isHovered = true
	if h.onMouseIn != nil {
		h.onMouseIn(event.Position.X, event.Position.Y)
	}
}

// MouseMoved is called when the mouse moves within the widget.
// When pointer lock is active, this is NOT called — deltas are delivered
// directly from the GLFW cursor callback via DeliverDelta/onDragged.
func (h *HoverableShader) MouseMoved(event *desktop.MouseEvent) {
	if h.onMouseMoved != nil {
		h.onMouseMoved(event.Position.X, event.Position.Y)
	}
}

// MouseOut is called when the mouse leaves the widget
func (h *HoverableShader) MouseOut() {
	h.isHovered = false
	if h.onMouseOut != nil {
		h.onMouseOut()
	}
}

// SetOnMouseIn sets the callback for mouse enter events
func (h *HoverableShader) SetOnMouseIn(callback MouseEventCallback) {
	h.onMouseIn = callback
}

// SetOnMouseMoved sets the callback for mouse move events

	if h.onMouseUp != nil {

}

// --- fyne.Focusable ---

// FocusGained is called when the widget gains keyboard focus
func (h *HoverableShader) FocusGained() {
	h.focused = true
}

// FocusLost is called when the widget loses keyboard focus
func (h *HoverableShader) FocusLost() {
	h.focused = false
	// Release all pressed keys
	for k := range h.keysPressed {
		if h.onKeyUp != nil {
			h.onKeyUp(k)
		}
		delete(h.keysPressed, k)
	}
}

// hoverableNormalizeKey lowercases single-character keys for consistency
func hoverableNormalizeKey(key string) string {
	if len(key) == 1 {
		return strings.ToLower(key)
	}
	return key
}

// TypedRune is called when a printable character is typed while focused
func (h *HoverableShader) TypedRune(r rune) {
	key := hoverableNormalizeKey(string(r))
	if !h.keysPressed[key] {
		h.keysPressed[key] = true
		if h.onKeyDown != nil {
			h.onKeyDown(key)
		}
	}
}

// TypedKey is called when a special key is typed while focused
func (h *HoverableShader) TypedKey(e *fyne.KeyEvent) {
	key := hoverableNormalizeKey(string(e.Name))
	if !h.keysPressed[key] {
		h.keysPressed[key] = true
		if h.onKeyDown != nil {
			h.onKeyDown(key)
		}
	}
}

// --- desktop.Keyable ---

// KeyDown handles desktop.Keyable key-down events.
// Most keys are handled via TypedRune/TypedKey, but Escape is handled here
// to ensure pointer lock release works even if TypedKey is intercepted.
func (h *HoverableShader) KeyDown(e *fyne.KeyEvent) {
	if e.Name == fyne.KeyEscape && h.pointerLocked {
		key := hoverableNormalizeKey(string(e.Name))
		if !h.keysPressed[key] {
			h.keysPressed[key] = true
			if h.onKeyDown != nil {
				h.onKeyDown(key)
			}
		}
	}
}

// KeyUp is called when a key is released
func (h *HoverableShader) KeyUp(e *fyne.KeyEvent) {
	key := hoverableNormalizeKey(string(e.Name))
	delete(h.keysPressed, key)
	if h.onKeyUp != nil {
		h.onKeyUp(key)
	}
}

// --- Pointer Lock ---

// SetPointerLock enables or disables pointer lock mode.
// When enabled, mouse deltas are delivered from the GLFW cursor callback
// via DeliverDelta, bypassing Fyne's normal event dispatch.
func (h *HoverableShader) SetPointerLock(locked bool) {
	h.pointerLocked = locked
}

// IsPointerLocked returns whether pointer lock mode is active.
func (h *HoverableShader) IsPointerLocked() bool {
	return h.pointerLocked
}

// DeliverDelta delivers a pointer-lock mouse delta to the onDragged callback.
// Called directly from the GLFW cursor position callback when pointer lock
// is active, bypassing Fyne's normal event system.
func (h *HoverableShader) DeliverDelta(dx, dy float32) {
	if h.onDragged != nil {
		h.onDragged(dx, dy)
	}
}
