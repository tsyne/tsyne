package canvas

import (
	"sync"
	"time"

	"fyne.io/fyne/v2"
)

// AttributeBuffer holds data for a single vertex attribute
type AttributeBuffer struct {
	Data       []float32 // The actual vertex data
	Size       int       // Number of components per vertex (2 for vec2, 3 for vec3, etc.)
	VBO        uint32    // GL vertex buffer object ID (managed by painter)
	Stride     int       // Byte stride between consecutive vertices (0 = tightly packed)
	Offset     int       // Byte offset into the buffer for this attribute
	Generation uint64    // Incremented when Data changes; used for upload caching
}

// RenderCommand represents a single render operation (uniform set or draw call)
// This allows immediate-mode GL patterns: set uniform → draw → set uniform → draw
type RenderCommand struct {
	Type      string      // "uniform", "drawElements", "drawArrays", "clear"
	Name      string      // Uniform name (for uniform commands)
	Value     interface{} // Uniform value or draw parameters
}

// DrawElementsParams holds parameters for a drawElements call
type DrawElementsParams struct {
	Mode             uint32
	Count            int
	Offset           int
	InstanceCount    int                         // 0 means non-instanced
	AttributeBuffers map[string]*AttributeBuffer // Snapshot of attribute buffers at draw time
	Indices          []uint16                    // Snapshot of index buffer at draw time
}

// UseProgramParams holds parameters for a useProgram command
type UseProgramParams struct {
	VertexSource   string
	FragmentSource string
}

// DrawArraysParams holds parameters for a drawArrays call
type DrawArraysParams struct {
	Mode             uint32
	First            int
	Count            int
	InstanceCount    int                         // 0 means non-instanced
	AttributeBuffers map[string]*AttributeBuffer // Snapshot of attribute buffers at draw time
}

// BindFramebufferParams holds parameters for a bindFramebuffer command
type BindFramebufferParams struct {
	Target        uint32
	FramebufferId int // JS-side ID (0 = default/unbind)
}

// FramebufferTexture2DParams holds parameters for a framebufferTexture2D command
type FramebufferTexture2DParams struct {
	Target     uint32
	Attachment uint32
	Textarget  uint32
	TextureId  int // JS-side texture ID
	Level      int
}

// RenderbufferStorageParams holds parameters for a renderbufferStorage command
type RenderbufferStorageParams struct {
	Target         uint32
	Internalformat uint32
	Width          int
	Height         int
}

// FramebufferRenderbufferParams holds parameters for a framebufferRenderbuffer command
type FramebufferRenderbufferParams struct {
	Target             uint32
	Attachment         uint32
	RenderbufferTarget uint32
	RenderbufferId     int
}

// ViewportParams holds parameters for a viewport command
type ViewportParams struct {
	X, Y, Width, Height int
}

// ColorMaskParams holds parameters for a colorMask command
type ColorMaskParams struct {
	R, G, B, A bool
}

// ClearDepthParams holds parameters for a clearDepth command
type ClearDepthParams struct {
	Depth float32
}

// DrawBuffersParams holds parameters for a drawBuffers command
type DrawBuffersParams struct {
	Buffers []uint32
}

// TexImage2DGPUParams holds parameters for a GPU-only texImage2D (null pixels)
type TexImage2DGPUParams struct {
	Target         uint32
	Level          int
	Internalformat uint32
	Width          int
	Height         int
	Format         uint32
	Type           uint32
}

// TexParameteriParams holds parameters for a texParameteri command
type TexParameteriParams struct {
	Target uint32
	Pname  uint32
	Param  int32
}

// Shader is a canvas object that renders using a custom GLSL vertex and fragment shader.
// This enables GPU-accelerated effects like fractals, ray marching, 3D geometry, etc.
type Shader struct {
	baseObject
	size            fyne.Size
	VertexSource    string                 // GLSL vertex shader source code (optional)
	FragmentSource  string                 // GLSL fragment shader source code
	Uniforms        map[string]interface{} // Uniform values: float32, [2]float32, [3]float32, [4]float32
	uniformsMu      sync.RWMutex           // Protects concurrent access to Uniforms map
	Textures        map[string]interface{} // Texture uniforms: image.Image, image.RGBA, etc.
	Cubemaps        map[string][6]interface{} // Cubemap uniforms: [+X, -X, +Y, -Y, +Z, -Z]

	// Vertex buffer data (legacy interleaved format)
	Vertices        []float32              // Vertex data (positions, normals, texcoords, etc.)
	Indices         []uint16               // Index buffer (for indexed drawing)
	VertexFormat    string                 // Format descriptor: "pos3", "pos3_norm3", "pos3_norm3_uv2", etc.

	// Separate attribute buffers (for three.js style rendering)
	AttributeBuffers map[string]*AttributeBuffer // Map of attribute name → buffer data
	attrBuffersMu    sync.RWMutex                // Protects concurrent access to AttributeBuffers

	// Time tracking for u_time uniform (only updates on Refresh, not every paint pass)
	startTime       time.Time              // When the shader was created
	cachedTime      float32                // Cached u_time value, only updated on Refresh()

	// Internal state managed by the GL painter
	program         uint32
	needsCompile    bool
	uniformLocs     map[string]int32
	textureUnits    map[string]int    // Mapping of texture uniform names to GL texture units
	textureCache    map[string]uint32 // Cache of GL texture IDs for regular textures
	cubemapUnits    map[string]int    // Mapping of cubemap names to GL texture units
	cubemapCache    map[string]uint32 // Cache of GL cubemap texture IDs
	vbo             uint32                 // Vertex buffer object ID (legacy)
	ibo             uint32                 // Index buffer object ID
	vertexCount     int                    // Number of vertices for rendering
	indexCount      int                    // Number of indices (if using indexed drawing)

	// Render command queue for immediate-mode GL patterns
	renderCommands      []RenderCommand
	lastRenderCommands  []RenderCommand // preserved from previous frame for replay
	renderMu            sync.Mutex
	LastGetWasNew       bool            // true if last GetRenderCommands returned new (not replay)

	// Paint synchronization - wait for Fyne to actually paint before next frame
	paintDone       chan struct{}
	paintPending    bool
	paintMu         sync.Mutex

	// Clear color for glClearColor
	clearColor      [4]float32

	// Attribute locations from JS side (bypasses OpenGL's glGetAttribLocation)
	// This is needed because OpenGL optimizes out unused attributes, but we
	// still need to bind the data to the expected locations
	attributeLocations map[string]int32
	attribLocMu        sync.RWMutex

	// Multi-program support for scenes with multiple shader programs
	// Maps source key → compiled GL program ID
	programCache    map[string]uint32
	// Pending program sources (set by useProgram command for painter to compile)
	pendingPrograms []PendingProgram
	pendingProgMu   sync.Mutex

	// FBO support for shadow mapping / offscreen rendering
	fboCache    map[int]uint32  // JS framebuffer ID → GL framebuffer ID
	rboCache    map[int]uint32  // JS renderbuffer ID → GL renderbuffer ID
	gpuTexCache      map[int]uint32       // JS texture ID → GL texture ID (for GPU-only textures)
	jsTexForSampler  map[string]int       // sampler uniform name → JS texture ID
	cpuTexImages     map[int]interface{}  // JS texture ID → CPU image (image.Image)
	gpuTexUploaded   map[int]bool         // JS texture ID → whether GPU texture has real data
	gpuTexFormats    map[int]GPUTextureFormat // JS texture ID → format info for upload

	// Instanced drawing
	attribDivisors   map[string]uint32    // attribute name → divisor value

	// VBO upload cache: tracks which generation was last uploaded to each VBO.
	// Key is VBO ID, value is the generation that was last uploaded.
	// The painter checks this before re-uploading: if generation matches, skip BufferData.
	VBOUploadedGen   map[uint32]uint64

	// Mat4Locs tracks GL locations reserved by mat4 attributes (reused across frames).
	Mat4Locs         map[int]bool

	// Capture-on-next-paint: when set, the painter reads pixels at the end of drawShader
	// (before SwapBuffers) and sends the result through the channel.
	captureChan      chan CaptureResult
	captureMu        sync.Mutex

	// readPixels support: JS requests pixel data, painter executes on GL thread
	readPixelsChan   chan ReadPixelsResult
	readPixelsReq    *ReadPixelsRequest
	readPixelsMu     sync.Mutex
}

// ReadPixelsRequest holds parameters for a pending readPixels call.
type ReadPixelsRequest struct {
	X, Y, Width, Height int
}

// ReadPixelsResult holds pixel data from glReadPixels.
type ReadPixelsResult struct {
	Pixels []uint8
}

// CaptureResult holds pixel data captured during a paint cycle.
type CaptureResult struct {
	Pixels        []uint8
	Width, Height int
}

// GPUTextureFormat stores the GL format info for a texture so the painter
// can upload with the correct format (e.g., GL_RED vs GL_RGBA).
type GPUTextureFormat struct {
	Internalformat uint32
	Format         uint32
	Type           uint32
}

// PendingProgram holds shader sources queued for compilation by the painter
type PendingProgram struct {
	VertexSource   string
	FragmentSource string
}

// NewShader creates a new shader canvas object with the given dimensions and fragment shader.
// The fragment shader should output to gl_FragColor. Available uniforms:
//   - vec2 u_resolution: canvas size in pixels
//   - float u_time: time in seconds since start
//   - vec2 u_mouse: mouse position (if provided via SetUniform)
// Custom uniforms can be set via SetUniform.
// Texture uniforms can be set via SetTextureUniform.
// Cubemap uniforms can be set via SetCubemapUniform.
// Vertex data can be set via SetVertices and SetIndices.
func NewShader(width, height float32, fragmentSrc string) *Shader {
	now := time.Now()
	s := &Shader{
		VertexSource:     "",
		FragmentSource:   fragmentSrc,
		Uniforms:         make(map[string]interface{}),
		Textures:         make(map[string]interface{}),
		Cubemaps:         make(map[string][6]interface{}),
		Vertices:         nil,
		Indices:          nil,
		VertexFormat:     "",
		AttributeBuffers: make(map[string]*AttributeBuffer),
		startTime:        now,
		cachedTime:       0, // Initial u_time is 0
		needsCompile:     true,
		uniformLocs:      make(map[string]int32),
		textureUnits:     make(map[string]int),
		textureCache:     make(map[string]uint32),
		cubemapUnits:     make(map[string]int),
		cubemapCache:     make(map[string]uint32),
		vbo:                0,
		ibo:                0,
		vertexCount:        0,
		indexCount:         0,
		attributeLocations: make(map[string]int32),
		programCache:       make(map[string]uint32),
		fboCache:           make(map[int]uint32),
		rboCache:           make(map[int]uint32),
		gpuTexCache:        make(map[int]uint32),
		jsTexForSampler:    make(map[string]int),
		cpuTexImages:       make(map[int]interface{}),
		gpuTexUploaded:     make(map[int]bool),
		gpuTexFormats:      make(map[int]GPUTextureFormat),
		attribDivisors:     make(map[string]uint32),
	}
	s.size = fyne.NewSize(width, height)
	return s
}

// Size returns the current size of the shader canvas
func (s *Shader) Size() fyne.Size {
	return s.size
}

// Refresh causes the shader to be redrawn and updates the cached u_time value.
// This is the ONLY time u_time changes - it does NOT update on every paint pass.
// This matches Fyne's texture caching behavior where widgets only regenerate on Refresh().
func (s *Shader) Refresh() {
	s.cachedTime = float32(time.Since(s.startTime).Seconds())
	Refresh(s)
}

// CachedTime returns the cached u_time value (only updated on Refresh)
func (s *Shader) CachedTime() float32 {
	return s.cachedTime
}

// StartTime returns when the shader was created
func (s *Shader) StartTime() time.Time {
	return s.startTime
}

// MinSize returns the minimum size (same as size for shaders)
func (s *Shader) MinSize() fyne.Size {
	return s.size
}

// Resize changes the shader canvas size
func (s *Shader) Resize(size fyne.Size) {
	s.size = size
	s.Refresh()
}

// SetMinSize sets the shader canvas size
func (s *Shader) SetMinSize(size fyne.Size) {
	s.size = size
}

// SetUniform sets a uniform value. Supported types:
//   - float32 or float64 -> float uniform
//   - [2]float32 or []float32 (len 2) -> vec2 uniform
//   - [3]float32 or []float32 (len 3) -> vec3 uniform
//   - [4]float32 or []float32 (len 4) -> vec4 uniform
//   - int or int32 -> int uniform
func (s *Shader) SetUniform(name string, value interface{}) {
	s.uniformsMu.Lock()
	s.Uniforms[name] = value
	s.uniformsMu.Unlock()
	s.Refresh()
}

// GetUniformsCopy returns a thread-safe copy of the uniforms map for iteration.
// Use this in the painter to avoid concurrent map access.
func (s *Shader) GetUniformsCopy() map[string]interface{} {
	s.uniformsMu.RLock()
	defer s.uniformsMu.RUnlock()
	copy := make(map[string]interface{}, len(s.Uniforms))
	for k, v := range s.Uniforms {
		copy[k] = v
	}
	return copy
}

// SetSource updates the fragment shader source and marks for recompilation
// Note: Does NOT call Refresh() - caller should call Refresh() after all data is set up
func (s *Shader) SetSource(src string) {
	s.FragmentSource = src
	s.needsCompile = true
}

// SetVertexSource updates the vertex shader source and marks for recompilation
// Note: Does NOT call Refresh() - caller should call Refresh() after all data is set up
func (s *Shader) SetVertexSource(src string) {
	s.VertexSource = src
	s.needsCompile = true
}

// Program returns the compiled GL program ID (for painter use)
func (s *Shader) Program() uint32 {
	return s.program
}

// SetProgram sets the compiled GL program ID (for painter use)
func (s *Shader) SetProgram(p uint32) {
	s.program = p
	s.needsCompile = false
}

// NeedsCompile returns whether the shader needs (re)compilation
func (s *Shader) NeedsCompile() bool {
	return s.needsCompile
}

// UniformLocs returns the uniform location cache (for painter use)
func (s *Shader) UniformLocs() map[string]int32 {
	return s.uniformLocs
}

// SetUniformLoc caches a uniform location (for painter use)
func (s *Shader) SetUniformLoc(name string, loc int32) {
	s.uniformLocs[name] = loc
}
