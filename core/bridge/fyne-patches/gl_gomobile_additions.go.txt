func (c *mobileContext) DrawBuffers(bufs []uint32) {
	enums := make([]gl.Enum, len(bufs))
	for i, b := range bufs {
		enums[i] = gl.Enum(b)
	}
	c.glContext.DrawBuffers(enums)
}

func (c *mobileContext) TexImage2DEmpty(target uint32, level int, internalformat uint32, width, height int, format, typ uint32) {
	c.glContext.TexImage2D(gl.Enum(target), level, int(internalformat), width, height, gl.Enum(format), gl.Enum(typ), nil)
}
